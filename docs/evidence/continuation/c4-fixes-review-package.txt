## onto-contract/src/main/java/cn/ontorepo/contract/GuardValidator.java
package cn.ontorepo.contract;

import com.fasterxml.jackson.databind.JsonNode;
import java.util.Set;

/** 逐项复核完整守卫集合；缺定义或缺执行绑定均阻断，不存在仅声明安全豁免。 */
public final class GuardValidator {
    private static final Set<String> TYPES = Set.of("STRING", "INTEGER", "DECIMAL", "BOOLEAN");
    private GuardValidator() {}

    public static void validate(JsonNode operation, JsonNode guardCfg) {
        String code = ContractFields.text(operation, "code");
        JsonNode guards = operation.path("assistant_tool").path("guard_attrs");
        if (guards.isMissingNode() || guards.isNull() || (guards.isArray() && guards.isEmpty()))
            throw new ContractProblem("MISSING_GUARD_ATTRS", code, "MISSING_GUARD_ATTRS");
        var names = ContractFields.strings(guards, "guard_attrs", true);
        for (String name : names) {
            JsonNode definition = guardCfg.path(name);
            if (!definition.isObject()) unresolved(code, name, "UNDEFINED");
            if (!TYPES.contains(definition.path("type").asText())) unresolved(code, name, "UNSUPPORTED_TYPE");
            if (!filterBinding(definition.path("filter_binding"))
                    && !authorizationRule(definition.path("authorization_rule")))
                unresolved(code, name, "NO_FILTER_BINDING");
        }
    }

    /** 仅接受可由查询翻译器参数化执行的表列和运算符，任意字符串不算绑定。 */
    private static boolean filterBinding(JsonNode binding) {
        return binding.isObject() && textualIdentifier(binding.path("table"))
                && textualIdentifier(binding.path("column"))
                && binding.path("operator").isTextual()
                && Set.of("IN", "EQ").contains(binding.path("operator").textValue());
    }

    /** 授权规则必须指向受认证身份属性，不能让调用参数替代主体身份。 */
    private static boolean authorizationRule(JsonNode rule) {
        return rule.isObject() && "PRINCIPAL_ATTRIBUTE".equals(rule.path("kind").textValue())
                && rule.path("attribute").isTextual()
                && GuardAttributeName.valid(rule.path("attribute").textValue());
    }

    /** 标识符必须来自 JSON 字符串，禁止 asText 将布尔值伪装成合法名称。 */
    private static boolean textualIdentifier(JsonNode value) {
        return value.isTextual() && identifier(value.textValue());
    }

    /** 数据源表列标识符规则；属性命名独立由GuardAttributeName维护，不能顺带扩张SQL名称。 */
    public static boolean identifier(String value) {
        return value != null && value.matches("[A-Za-z_][A-Za-z0-9_]{0,127}");
    }

    private static void unresolved(String op, String attr, String reason) {
        throw new ContractProblem("GUARD_ATTR_UNRESOLVED", op + "." + attr, reason);
    }
}

## onto-data/src/main/java/cn/ontorepo/data/domain/LabelScope.java
package cn.ontorepo.data.domain;

import cn.ontorepo.contract.GuardAttributeName;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.node.DecimalNode;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

/**
 * 授权标签范围的不可变交集，不负责判断授权记录是否存在。
 *
 * <p>调用者必须先验证入口授权记录；记录存在且范围 null 才代表不限。 策略、端点、空间按属性逐项求交，任何层不能扩大另一层；同一空间范围应用两次保持等价。
 * 缺少分类标签始终拒绝，即使所有范围均为不限，也不能把未分类对象当作公共数据。
 */
public record LabelScope(Map<String, Set<JsonNode>> clauses) {
  public LabelScope {
    if (clauses == null || clauses.size() > 64) throw invalid();
    Map<String, Set<JsonNode>> copy = new LinkedHashMap<>();
    clauses.forEach(
        (attribute, values) -> {
          if (attribute == null
              || !GuardAttributeName.valid(attribute)
              || values == null
              || values.size() > 1000) throw invalid();
          Set<JsonNode> normalized = new LinkedHashSet<>();
          for (JsonNode value : values) {
            if (!classified(value)) throw invalid();
            normalized.add(normalize(value));
          }
          copy.put(attribute, Set.copyOf(normalized));
        });
    clauses = Map.copyOf(copy);
  }

  public static LabelScope unrestricted() {
    return new LabelScope(Map.of());
  }

  /** 仅允许明确的集合或集合合取，不接受原始 SQL、通配符和任意表达式。 */
  public static LabelScope parse(JsonNode source) {
    return parse(source, 0);
  }

  private static LabelScope parse(JsonNode source, int depth) {
    if (source == null || source.isNull()) return unrestricted();
    if (!source.isObject() || depth > 16) throw invalid();
    if (source.size() == 1 && source.has("all")) {
      JsonNode all = source.get("all");
      if (!all.isArray() || all.isEmpty() || all.size() > 64) throw invalid();
      LabelScope combined = unrestricted();
      for (JsonNode child : all) {
        if (!child.isObject()) throw invalid();
        combined = intersect(combined, parse(child, depth + 1));
      }
      return combined;
    }
    JsonNode attr = source.get("attr"), values = source.get("in");
    if (source.size() != 2
        || attr == null
        || !attr.isTextual()
        || !GuardAttributeName.valid(attr.textValue())
        || values == null
        || !values.isArray()
        || values.size() > 1000) throw invalid();
    Set<JsonNode> allowed = new LinkedHashSet<>();
    for (JsonNode value : values) {
      if (!classified(value)) throw invalid();
      allowed.add(normalize(value));
    }
    return new LabelScope(Map.of(attr.textValue(), allowed));
  }

  public static LabelScope intersect(LabelScope... scopes) {
    Map<String, Set<JsonNode>> result = new LinkedHashMap<>();
    for (LabelScope scope : scopes) {
      if (scope == null) throw invalid();
      for (var entry : scope.clauses.entrySet()) {
        Set<JsonNode> existing = result.get(entry.getKey());
        if (existing == null) result.put(entry.getKey(), new LinkedHashSet<>(entry.getValue()));
        else existing.retainAll(entry.getValue());
      }
    }
    return new LabelScope(result);
  }

  /** SQL 计划器将空交集编译为恒假谓词，不能在空列表时省略 WHERE。 */
  public boolean deniesAll() {
    return clauses.values().stream().anyMatch(Set::isEmpty);
  }

  /** 只判断当前属性；整条记录必须对全部必需 guard_attrs 逐项调用。 */
  public boolean allows(String attribute, JsonNode label) {
    if (deniesAll() || !classified(label)) return false;
    Set<JsonNode> allowed = clauses.get(attribute);
    return allowed == null || allowed.contains(normalize(label));
  }

  /** 返回参数化计划所需的只读集合；不得将其中字符串拼接到 SQL。 */
  public Map<String, List<JsonNode>> parameters() {
    Map<String, List<JsonNode>> result = new LinkedHashMap<>();
    clauses.forEach((attribute, values) -> result.put(attribute, List.copyOf(values)));
    return Map.copyOf(result);
  }

  private static JsonNode normalize(JsonNode value) {
    return value.isNumber()
        ? DecimalNode.valueOf(value.decimalValue().stripTrailingZeros())
        : value;
  }

  private static boolean classified(JsonNode value) {
    return value != null
        && (value.isBoolean()
            || value.isNumber()
            || value.isTextual()
                && !value.textValue().isBlank()
                && value.textValue().length() <= 1000);
  }

  private static DataProblem invalid() {
    return new DataProblem("INVALID_DATA_SCOPE", "数据范围必须为明确的属性值集合或集合合取");
  }
}


## onto-data/src/main/java/cn/ontorepo/data/domain/QueryPlanner.java
package cn.ontorepo.data.domain;

import cn.ontorepo.contract.GuardAttributeName;

import com.fasterxml.jackson.databind.JsonNode;
import java.util.*;
import java.util.stream.Collectors;

/** 全部源表和列先校验，标签参数化并在 LIMIT 前执行。 */
public final class QueryPlanner {
  public record SqlPlan(
      String sql,
      List<Object> parameters,
      int limit,
      int timeoutMs,
      List<String> projection,
      Map<String, List<String>> keys) {}

  public static void identifier(String name) {
    if (name == null
        || !name.matches("[A-Za-z_][A-Za-z0-9_]{0,79}(\\.[A-Za-z_][A-Za-z0-9_]{0,79}){0,2}"))
      throw problem("INVALID_IDENTIFIER");
  }

  public static DataProblem problem(String code) {
    return new DataProblem(code, "数据源查询不满足访问边界或能力约束");
  }

  public static Object scalar(JsonNode n) {
    if (n == null || n.isNull() || n.isContainerNode()) throw problem("INVALID_FILTER");
    if (n.isNumber()) return n.decimalValue();
    if (n.isBoolean()) return n.booleanValue();
    return n.textValue();
  }

  public SqlPlan plan(SourceSpec source, QuerySpec q, SourceMetadata metadata) {
    if (!q.options().isEmpty()) throw problem("UNSUPPORTED_OPTION");
    Map<String, SourceMetadata.Resource> resources =
        metadata.resources().stream()
            .collect(Collectors.toMap(SourceMetadata.Resource::name, r -> r));
    Set<String> used = new LinkedHashSet<>();
    used.add(q.resource());
    q.joins().forEach(j -> used.add(j.resource()));
    for (String r : used) {
      identifier(r);
      if (!source.accessScope().contains(r)) throw problem("SCOPE_EXCEEDED");
      if (!resources.containsKey(r)) throw problem("RESOURCE_NOT_FOUND");
    }
    Set<String> columns = new HashSet<>();
    used.forEach(
        r ->
            resources
                .get(r)
                .fields()
                .forEach(
                    f -> {
                      identifier(f.name());
                      columns.add(r + "." + f.name());
                    }));
    java.util.function.Function<String, String> col =
        c -> {
          identifier(c);
          if (!columns.contains(c)) throw problem("UNKNOWN_COLUMN");
          return quote(c);
        };
    if (q.guardBindings().isEmpty()
        || !q.guardBindings().keySet().stream()
            .map(QueryPlanner::guardAttribute)
            .collect(Collectors.toSet())
            .containsAll(q.scope().clauses().keySet())) throw problem("GUARD_ATTR_UNRESOLVED");
    // JOIN 每一端都需分类标签；不能通过已授权任务关联出未分类算法。
    for (String resource : used)
      if (q.guardBindings().values().stream().noneMatch(c -> c.startsWith(resource + ".")))
        throw problem("GUARD_ATTR_UNRESOLVED");
    List<Object> params = new ArrayList<>();
    List<String> where = new ArrayList<>();
    for (var e : q.guardBindings().entrySet()) {
      String c = col.apply(e.getValue());
      where.add(c + " IS NOT NULL");
      var field =
          resources.get(e.getValue().substring(0, e.getValue().lastIndexOf('.'))).fields().stream()
              .filter(
                  f -> f.name().equals(e.getValue().substring(e.getValue().lastIndexOf('.') + 1)))
              .findFirst()
              .orElseThrow();
      if (field.type().toUpperCase(Locale.ROOT).matches(".*(CHAR|TEXT|STRING).*"))
        where.add(classifiedText(source, c));
      var values = q.scope().parameters().get(guardAttribute(e.getKey()));
      if (values != null) in(c, values, where, params);
    }
    for (var f : q.filters()) {
      String c = col.apply(f.column());
      if (f.operator() == null || f.values().size() > 1000) throw problem("INVALID_FILTER");
      if (f.operator() == QuerySpec.Operator.IN) in(c, f.values(), where, params);
      else {
        if (f.values().size() != 1) throw problem("INVALID_FILTER");
        where.add(
            c
                + " "
                + switch (f.operator()) {
                  case EQ -> "=";
                  case GT -> ">";
                  case GE -> ">=";
                  case LT -> "<";
                  case LE -> "<=";
                  default -> throw problem("INVALID_FILTER");
                }
                + " ?");
        params.add(scalar(f.values().getFirst()));
      }
    }
    String select = q.projection().stream().map(col).collect(Collectors.joining(","));
    Map<String, List<String>> keys = new LinkedHashMap<>();
    int keyIndex = 0;
    for (String r : used) {
      if (resources.get(r).primaryKey().isEmpty()) throw problem("SOURCE_KEY_REQUIRED");
      keys.put(r, resources.get(r).primaryKey());
      for (String key : resources.get(r).primaryKey())
        select += "," + col.apply(r + "." + key) + " AS \"__e" + (keyIndex++) + "\"";
    }
    StringBuilder sql =
        new StringBuilder("SELECT ").append(select).append(" FROM ").append(quote(q.resource()));
    Set<String> joined = new HashSet<>();
    joined.add(q.resource());
    for (var j : q.joins()) {
      String l = j.leftColumn(), r = j.rightColumn();
      if (!r.startsWith(j.resource() + ".")
          || joined.stream().noneMatch(x -> l.startsWith(x + "."))
          || !joined.add(j.resource())) throw problem("INVALID_JOIN");
      sql.append(" JOIN ")
          .append(quote(j.resource()))
          .append(" ON ")
          .append(col.apply(l))
          .append(" = ")
          .append(col.apply(r));
    }
    sql.append(" WHERE ").append(String.join(" AND ", where)).append(" LIMIT ?");
    params.add(q.limit());
    return new SqlPlan(
        sql.toString(),
        List.copyOf(params),
        q.limit(),
        q.timeoutMs(),
        q.projection(),
        Collections.unmodifiableMap(new LinkedHashMap<>(keys)));
  }

  private static void in(
      String col, List<JsonNode> values, List<String> where, List<Object> params) {
    if (values.isEmpty()) {
      where.add("1=0");
      return;
    }
    where.add(col + " IN (" + String.join(",", Collections.nCopies(values.size(), "?")) + ")");
    values.forEach(v -> params.add(scalar(v)));
  }

  public static String quote(String value) {
    identifier(value);
    return Arrays.stream(value.split("\\."))
        .map(x -> "\"" + x + "\"")
        .collect(Collectors.joining("."));
  }

  /** 空白集合从当前JDK规则取得，避免SQL默认TRIM仅删除普通空格。没有调用者输入进入字面量。 */
  private static String classifiedText(SourceSpec source, String column) {
    String dialect = source.connection().path("dialect").asText();
    StringBuilder escaped = new StringBuilder(), pattern = new StringBuilder("[");
    for (int point : BlankCharacters.POINTS) {
      escaped
          .append('\\')
          .append(
              point <= 0xffff
                  ? String.format(Locale.ROOT, "%04x", point)
                  : String.format(Locale.ROOT, "+%06x", point));
      pattern.append("\\\\x{").append(Integer.toHexString(point)).append('}');
    }
    pattern.append(']');
    return switch (dialect) {
      case "postgresql" -> "BTRIM(" + column + ", U&'" + escaped + "') <> ''";
      case "clickhouse" -> "replaceRegexpAll(" + column + ", '" + pattern + "', '') <> ''";
      default -> throw problem("UNSUPPORTED_DIALECT");
    };
  }

  private static final class BlankCharacters {
    static final int[] POINTS =
        java.util.stream.IntStream.rangeClosed(0, Character.MAX_CODE_POINT)
            .filter(Character::isWhitespace)
            .toArray();
  }

  /** 同一属性在 JOIN 多表绑定时使用 attr@resource，仍应用同一属性范围。 */
  public static String guardAttribute(String binding) {
    String[] parts = binding.split("@", -1);
    if (parts.length > 2 || !GuardAttributeName.valid(parts[0]))
      throw problem("GUARD_ATTR_UNRESOLVED");
    if (parts.length == 2) identifier(parts[1]);
    return parts[0];
  }
}

package cn.ontorepo.data.domain;

import static org.junit.jupiter.api.Assertions.*;

import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.*;
import org.junit.jupiter.api.Test;

class QueryPlannerTest {
  private final ObjectMapper json = new ObjectMapper();
  private final SourceSpec source =
      new SourceSpec(
          "s",
          "RELATIONAL",
          json.createObjectNode().put("dialect", "postgresql"),
          Set.of("public.tasks", "public.algorithms"));
  private final SourceMetadata metadata =
      new SourceMetadata(
          List.of(
              new SourceMetadata.Resource(
                  "public.tasks",
                  List.of(
                      new SourceMetadata.Field("id", "INTEGER", false),
                      new SourceMetadata.Field("region", "VARCHAR", true),
                      new SourceMetadata.Field("algo", "INTEGER", false)),
                  List.of("id"),
                  List.of(),
                  List.of()),
              new SourceMetadata.Resource(
                  "public.algorithms",
                  List.of(
                      new SourceMetadata.Field("id", "INTEGER", false),
                      new SourceMetadata.Field("region", "VARCHAR", true)),
                  List.of("id"),
                  List.of(),
                  List.of())));

  QuerySpec query(LabelScope scope, Map<String, String> guards) {
    return new QuerySpec(
        "public.tasks",
        List.of("public.tasks.id"),
        List.of(),
        List.of(),
        10000,
        90000,
        Map.of(),
        scope,
        guards);
  }

  @Test
  void historicalAttributeNamesKeepQualifiedBindingsAndSourceIdentifierBoundaries() {
    for(String attribute:List.of("_region","r".repeat(128))){
      var scope=LabelScope.parse(json.valueToTree(Map.of("attr",attribute,"in",List.of("A"))));
      var plan=assertDoesNotThrow(()->new QueryPlanner().plan(source,query(scope,Map.of(attribute+"@public.tasks","public.tasks.region")),metadata));
      assertEquals(List.of("A",500),plan.parameters());
      assertTrue(plan.sql().contains("IS NOT NULL"));
      assertThrows(DataProblem.class,()->new QueryPlanner().plan(source,query(scope,Map.of(attribute+"@public.tasks;DROP","public.tasks.region")),metadata));
    }
  }

  @Test
  void bindsLabelsAndCapsExecution() throws Exception {
    var scope = LabelScope.parse(json.readTree("{\"attr\":\"region\",\"in\":[\"A' OR 1=1 --\"]}"));
    var plan =
        new QueryPlanner()
            .plan(source, query(scope, Map.of("region", "public.tasks.region")), metadata);
    assertFalse(plan.sql().contains("OR 1=1"));
    assertTrue(plan.sql().contains("IS NOT NULL"));
    assertEquals("A' OR 1=1 --", plan.parameters().getFirst());
    assertEquals(500, plan.limit());
    assertEquals(10000, plan.timeoutMs());
  }

  @Test
  void emptyIntersectionIsFalseAndMissingGuardFails() {
    var scope = new LabelScope(Map.of("region", Set.of()));
    assertTrue(
        new QueryPlanner()
            .plan(source, query(scope, Map.of("region", "public.tasks.region")), metadata)
            .sql()
            .contains("1=0"));
    assertThrows(
        DataProblem.class, () -> new QueryPlanner().plan(source, query(scope, Map.of()), metadata));
    assertThrows(
        DataProblem.class,
        () ->
            new QueryPlanner().plan(source, query(LabelScope.unrestricted(), Map.of()), metadata));
  }

  @Test
  void rejectsUnknownTablesColumnsAndOptions() {
    assertThrows(
        DataProblem.class,
        () ->
            new QueryPlanner()
                .plan(
                    source,
                    query(LabelScope.unrestricted(), Map.of("region", "public.tasks.bad")),
                    metadata));
    assertThrows(
        DataProblem.class,
        () ->
            new QueryPlanner()
                .plan(
                    new SourceSpec("s", "RELATIONAL", source.connection(), Set.of()),
                    query(LabelScope.unrestricted(), Map.of("region", "public.tasks.region")),
                    metadata));
    assertThrows(
        DataProblem.class,
        () ->
            new QueryPlanner()
                .plan(
                    source,
                    new QuerySpec(
                        "public.tasks;DROP",
                        List.of("id"),
                        List.of(),
                        List.of(),
                        1,
                        1,
                        Map.of(),
                        LabelScope.unrestricted(),
                        Map.of("region", "region")),
                    metadata));
  }

  @Test
  void joinsRequireGuardForEveryParticipatingTable() {
    var joined =
        new QuerySpec(
            "public.tasks",
            List.of("public.tasks.id", "public.algorithms.id"),
            List.of(),
            List.of(
                new QuerySpec.Join(
                    "public.algorithms", "public.tasks.algo", "public.algorithms.id")),
            1,
            1000,
            Map.of(),
            LabelScope.unrestricted(),
            Map.of("region", "public.tasks.region"));
    assertEquals(
        "GUARD_ATTR_UNRESOLVED",
        assertThrows(DataProblem.class, () -> new QueryPlanner().plan(source, joined, metadata))
            .code());
    var complete =
        new QuerySpec(
            joined.resource(),
            joined.projection(),
            joined.filters(),
            joined.joins(),
            1,
            1000,
            Map.of(),
            LabelScope.unrestricted(),
            Map.of(
                "region",
                "public.tasks.region",
                "region@public.algorithms",
                "public.algorithms.region"));
    assertTrue(new QueryPlanner().plan(source, complete, metadata).sql().contains("JOIN"));
  }
}

/** 空间守卫定义属于草稿，授权属性属于实时主体；两者不能互相代替。 */
@Service
public class GuardService {
    private final Db db;
    private final SpaceService spaces;
    private static final ObjectMapper JSON = new ObjectMapper();

    public GuardService(Db db, SpaceService spaces) {
        this.db = db;
        this.spaces = spaces;
    }

    private void owner(UUID u, UUID s) {
        if (!"OWNER".equals(spaces.role(u, s)))
            throw new Problem(403, "FORBIDDEN", "仅OWNER可管理守卫和授权");
    }

    public Map<String, Object> guards(UUID u, UUID s) {
        spaces.role(u, s);
        return Map.of(
                "guardCfg",
                db.one("SELECT guard_cfg FROM ontology_space WHERE id=?", s).get("guardCfg"));
    }

    @Transactional
    public Map<String, Object> saveGuards(UUID u, UUID s, Map<String, Object> b) {
        return spaces.write(
                u,
                s,
                Input.number(b, "expectedDraftRev"),
                "GUARD_SAVE",
                () -> {
                    owner(u, s);
                    JsonNode cfg = JSON.valueToTree(b.get("guardCfg"));
                    if (!cfg.isObject() || cfg.size() > 64) throw Problem.bad("守卫配置必须为最多64项的对象");
                    cfg.fields()
                            .forEachRemaining(
                                    e -> {
                                        JsonNode d = e.getValue();
                                        if (!GuardAttributeName.valid(e.getKey())
                                                || !d.isObject()
                                                || !Set.of(
                                                                "STRING", "INTEGER", "DECIMAL",
                                                                "BOOLEAN")
                                                        .contains(d.path("type").asText()))
                                            throw Problem.bad("守卫名称或类型无效");
                                        // 草稿允许缺绑定，但显式提交的绑定必须结构正确，不能把任意代码当作绑定。
                                        if (d.has("authorization_rule")) {
                                            var r = d.path("authorization_rule");
                                            if (!r.isObject()
                                                    || !"PRINCIPAL_ATTRIBUTE"
                                                            .equals(r.path("kind").asText())
                                                    || !r.path("attribute").isTextual()
                                                    || !GuardAttributeName.valid(
                                                            r.path("attribute").asText()))
                                                throw Problem.bad("主体属性绑定无效");
                                        }
                                        if (d.has("filter_binding")) {
                                            var f = d.path("filter_binding");
                                            if (!f.isObject()
                                                    || !f.path("table").isTextual()
                                                    || !f.path("column").isTextual()
                                                    || !GuardValidator.identifier(
                                                            f.path("table").asText())
                                                    || !GuardValidator.identifier(
                                                            f.path("column").asText())
                                                    || !Set.of("IN", "EQ")
                                                            .contains(f.path("operator").asText()))
                                                throw Problem.bad("数据过滤绑定无效");
                                        }
                                    });
                    db.update(
                            "UPDATE ontology_space SET guard_cfg=?::jsonb WHERE id=?",
                            db.json(b.get("guardCfg")),
                            s);
                });
    }

    public List<Map<String, Object>> grants(UUID u, UUID s) {
        owner(u, s);
        return db.list(
                "SELECT g.user_id,u.username,u.display_name,g.role,g.data_scope,g.attributes FROM"
                    + " space_grant g JOIN local_user u ON u.id=g.user_id WHERE g.space_id=? ORDER"
                    + " BY u.username",
                s);
    }

    public List<Map<String, Object>> users(UUID u, UUID s) {
        owner(u, s);
        return db.list("SELECT id user_id,username,display_name FROM local_user ORDER BY username");
    }

    @Transactional
    public Map<String, Object> saveGrant(UUID u, UUID s, UUID target, Map<String, Object> b) {
        return spaces.write(
                u,
                s,
                Input.number(b, "expectedDraftRev"),
                "GRANT_SAVE",
                () -> {
                    owner(u, s);
                    db.one("SELECT id FROM local_user WHERE id=?", target);
                    String role = Input.text(b, "role", 10);
                    if (!Set.of("OWNER", "EDITOR", "VIEWER").contains(role))
                        throw Problem.bad("授权角色无效");
                    if (!role.equals("OWNER")
                            && !db.list(
                                            "SELECT 1 FROM space_grant WHERE space_id=? AND"
                                                    + " user_id=? AND role='OWNER'",
                                            s,
                                            target)
                                    .isEmpty()
                            && db.list(
                                                    "SELECT 1 FROM space_grant WHERE space_id=? AND"
                                                            + " role='OWNER'",
                                                    s)
                                            .size()
                                    == 1) throw new Problem(409, "LAST_OWNER", "不能移除最后一个OWNER");
                    JsonNode attrs = JSON.valueToTree(b.getOrDefault("attributes", Map.of()));
                    if (!attrs.isObject() || attrs.size() > 64) throw Problem.bad("主体属性必须为对象");
                    JsonNode cfg =
                            JSON.valueToTree(
                                    db.one("SELECT guard_cfg FROM ontology_space WHERE id=?", s)
                                            .get("guardCfg"));
                    attrs.fields()
                            .forEachRemaining(
                                    e -> {
                                        if (!GuardAttributeName.valid(e.getKey())
                                                || !e.getValue().isValueNode()
                                                || e.getValue().isNull()
                                                || e.getValue().isTextual()
                                                        && e.getValue().asText().isBlank())
                                            throw Problem.bad("主体属性必须是非空标量");
                                    });
                    cfg.fields()
                            .forEachRemaining(
                                    e -> {
                                        JsonNode d = e.getValue();
                                        String attr =
                                                d.path("authorization_rule")
                                                        .path("attribute")
                                                        .asText();
                                        if (attrs.has(attr))
                                            typed(d.path("type").asText(), attrs.path(attr));
                                    });
                    LabelScope.parse(JSON.valueToTree(b.get("dataScope")));
                    db.update(
                            "INSERT INTO space_grant(space_id,user_id,role,data_scope,attributes)"
                                + " VALUES(?,?,?,?::jsonb,?::jsonb) ON CONFLICT(space_id,user_id)"
                                + " DO UPDATE SET"
                                + " role=excluded.role,data_scope=excluded.data_scope,attributes=excluded.attributes",
                            s,
                            target,
                            role,
                            b.get("dataScope") == null ? null : db.json(b.get("dataScope")),
                            db.json(b.getOrDefault("attributes", Map.of())));
                });
    }

    /** 只有空间授权行存在时null范围才不限；读取发生在每次调用的空间锁内。 */
    public Map<String, Object> grant(UUID u, UUID s) {
        spaces.editor(u, s);
        return db.one(
                "SELECT data_scope,attributes FROM space_grant WHERE space_id=? AND user_id=?",
                s,
                u);
    }

    /** 严格JSON标量类型检查，禁止字符串数字或布尔值冒充受信属性。 */
    public static void typed(String type, JsonNode v) {
        boolean valid =
                switch (type) {
                    case "STRING" -> v.isTextual() && !v.asText().isBlank();
                    case "INTEGER" -> v.isIntegralNumber();
                    case "DECIMAL" -> v.isNumber();
                    case "BOOLEAN" -> v.isBoolean();
                    default -> false;
                };
        if (!valid) throw new Problem(403, "GUARD_ATTRIBUTE_INVALID", "主体守卫属性缺失或类型错误");
    }
}

package cn.ontorepo.trace.application;

import cn.ontorepo.common.domain.*;
import cn.ontorepo.knowledge.application.FrozenKnowledge;

import org.springframework.stereotype.Component;

import java.util.*;

/** 确定性NL选择：仅冻结声明、示例与词典可参与打分，不接收期望operationId。 */
@Component
public class TermMatcher {
    public UUID select(Map<String, Object> p, String question) {
        String q = normalize(question);
        if (q.isEmpty()) throw new Problem(400, "NL_OPERATION_NOT_FOUND", "问题未命中冻结操作声明");
        int best = 0;
        List<Map<String, Object>> found = new ArrayList<>();
        for (var op : FrozenKnowledge.rows(p, "operations")) {
            if (!(op.get("assistantTool") instanceof Map<?, ?> tool)) continue;
            int score = 0;
            for (Object text :
                    List.of(
                            op.get("name"),
                            op.get("code"),
                            Objects.toString(tool.get("name"), ""),
                            Objects.toString(tool.get("description"), ""))) {
                String t = normalize(text.toString());
                if (!t.isEmpty() && (q.contains(t) || t.contains(q)))
                    score = Math.max(score, t.equals(q) ? 100 : 20);
            }
            String declaration =
                    normalize(
                            op.get("name")
                                    + " "
                                    + tool.get("name")
                                    + " "
                                    + tool.get("description"));
            for (var term : FrozenKnowledge.rows(p, "knowledge")) {
                Set<String> names = new HashSet<>();
                names.add(normalize(Objects.toString(term.get("term"), "")));
                for (Object alias : (List<?>) term.getOrDefault("aliases", List.of()))
                    names.add(normalize(Objects.toString(alias, "")));
                // 已冻结的历史词条也可能只有标点；空token绝不能利用contains("")制造命中。
                names.remove("");
                boolean questionHit = names.stream().anyMatch(q::contains);
                boolean operationHit = names.stream().anyMatch(declaration::contains);
                if (questionHit && operationHit) score += 10;
            }
            if (tool.get("param_examples") instanceof List<?> samples)
                for (Object sample : samples) {
                    if (sample instanceof Map<?, ?> example
                            && example.get("question") instanceof String text
                            && q.equals(normalize(text))) score += 100;
                    Set<String> tokens = new HashSet<>();
                    exampleTokens(sample, tokens, 0);
                    if (tokens.stream().anyMatch(q::contains)) score += 5;
                }
            if (score > best) {
                best = score;
                found.clear();
                found.add(op);
            } else if (score > 0 && score == best) found.add(op);
        }
        if (found.isEmpty()) throw new Problem(400, "NL_OPERATION_NOT_FOUND", "问题未命中冻结操作声明");
        if (found.size() != 1)
            throw new Problem(400, "NL_OPERATION_AMBIGUOUS", "问题命中多个冻结操作，请补充限定条件");
        return Input.uuid(found.getFirst().get("id"));
    }

    private String normalize(String s) {
        return s.toLowerCase(Locale.ROOT).replaceAll("[\\s\\p{Punct}，。？！]", "");
    }

    /** 普通参数例中的稳定字符串也参与匹配，不能只识别非标准question字段。 */
    private void exampleTokens(Object value, Set<String> tokens, int depth) {
        if (depth > 32) return;
        if (value instanceof String text) {
            String token = normalize(text);
            if (token.length() >= 2) tokens.add(token);
        } else if (value instanceof Map<?, ?> map)
            for (Object child : map.values()) exampleTokens(child, tokens, depth + 1);
        else if (value instanceof List<?> list)
            for (Object child : list) exampleTokens(child, tokens, depth + 1);
    }
}

package cn.ontorepo.trace.application;

import cn.ontorepo.common.domain.*;
import cn.ontorepo.model.domain.PropertyFacets;
import cn.ontorepo.model.infra.RdfCodec;
import cn.ontorepo.operation.domain.*;

import org.apache.jena.query.*;
import org.apache.jena.rdf.model.*;
import org.springframework.stereotype.Component;

import java.util.*;
import java.util.concurrent.TimeUnit;

/** 执行仅接收冻结包；策略上限取交集，调用参数不能扩大资源边界。 */
@Component
public class FrozenExecutor {
    public record Result(
            List<String> columns,
            List<Map<String, Object>> rows,
            List<Map<String, String>> steps) {}

    private Map<String, String> step(String k, String d) {
        return Map.of("kind", k, "status", "SUCCEEDED", "detail", d);
    }

    @SuppressWarnings("unchecked")
    public Result execute(Map<String, Object> p, Map<String, Object> op, Object input) {
        var effective = FrozenPolicy.of(p, op);
        int max = effective.maxRows(), timeout = effective.timeoutMs();
        List<Map<String, String>> steps = new ArrayList<>();
        if (op.get("assistantTool") == null && !effective.scopes().isEmpty())
            throw new Problem(409, "READ_SCOPE_UNSUPPORTED", "当前MODEL_ONLY模型查询不能证明该读取范围");
        String kind = op.getOrDefault("kind", "OBJECT_QUERY").toString();
        List<String> columns = new ArrayList<>();
        List<Map<String, Object>> rows = new ArrayList<>();
        long started = System.nanoTime();
        if (kind.equals("OBJECT_QUERY") || kind.equals("RELATION_QUERY")) {
            Query query =
                    QueryPolicy.validate(
                            op.get("queryTemplate").toString(), max, Math.max(100, timeout));
            steps.add(step("VALIDATE_QUERY", "受限SELECT安全校验通过"));
            Model model = RdfCodec.model(p);
            try (QueryExecution qe =
                    QueryExecution.model(model)
                            .query(query)
                            .timeout(timeout, TimeUnit.MILLISECONDS)
                            .build()) {
                qe.getContext().set(ARQ.enablePropertyFunctions, false);
                ResultSet result = qe.execSelect();
                columns.addAll(result.getResultVars());
                while (result.hasNext() && rows.size() < max) {
                    QuerySolution solution = result.next();
                    Map<String, Object> row = new LinkedHashMap<>();
                    for (String col : columns) {
                        RDFNode value = solution.get(col);
                        row.put(
                                col,
                                value == null
                                        ? null
                                        : value.isLiteral()
                                                ? value.asLiteral().getLexicalForm()
                                                : value.toString());
                    }
                    rows.add(row);
                }
            } finally {
                model.close();
            }
            steps.add(step("MODEL_SELECT", "本地冻结模型查询完成，返回 " + rows.size() + " 行"));
        } else {
            if (!(input instanceof Map<?, ?> attributes))
                throw new Problem(400, "INPUT_ROOT_NOT_APPLICABLE", "规则和评分操作需要属性对象输入");
            Map<String, String> types = new HashMap<>();
            for (var prop : RdfCodec.rows(p, "properties"))
                types.put(prop.get("id").toString(), prop.get("rangeType").toString());
            Map<String, Object> canonicalInput = new LinkedHashMap<>();
            for (var entry : attributes.entrySet()) {
                String key = Input.uuid(entry.getKey()).toString();
                if (canonicalInput.containsKey(key)) throw Problem.bad("输入属性UUID重复");
                canonicalInput.put(key, entry.getValue());
            }
            for (var entry : canonicalInput.entrySet()) {
                if (!types.containsKey(entry.getKey())) throw Problem.bad("输入引用未知属性");
                if (entry.getValue() != null)
                    PropertyFacets.scalar(types.get(entry.getKey()), entry.getValue());
            }
            columns.addAll(
                    kind.equals("SCORING")
                            ? List.of("ruleId", "score", "explanation")
                            : List.of("ruleId", "passed", "message"));
            List<?> ids = (List<?>) op.getOrDefault("ruleIds", List.of());
            for (Object id : ids) {
                var rule =
                        RdfCodec.rows(p, "rules").stream()
                                .filter(r -> Input.sameUuid(r.get("id"), id))
                                .findFirst()
                                .orElseThrow(
                                        () ->
                                                new Problem(
                                                        409,
                                                        "FROZEN_REFERENCE_INVALID",
                                                        "冻结操作规则引用缺失"));
                if (RuleExpression.evaluate(rule.get("condition").toString(), canonicalInput)) {
                    Map<String, Object> row = new LinkedHashMap<>();
                    row.put("ruleId", rule.get("id").toString());
                    if (kind.equals("SCORING")) {
                        Map<String, Object> scoring = (Map<String, Object>) rule.get("scoring");
                        row.put(
                                "score",
                                ScoringPolicy.score(scoring, canonicalInput)
                                        .stripTrailingZeros()
                                        .toPlainString());
                        row.put(
                                "explanation",
                                Boolean.TRUE.equals(scoring.get("explain")) ? "按冻结指标、权重与分段计算" : "");
                    } else {
                        Map<String, Object> action = (Map<String, Object>) rule.get("action");
                        row.put("passed", Boolean.toString("ACCEPT".equals(action.get("type"))));
                        row.put("message", Objects.toString(action.get("message"), ""));
                    }
                    rows.add(row);
                }
                if (TimeUnit.NANOSECONDS.toMillis(System.nanoTime() - started) > timeout)
                    throw new QueryCancelledException();
            }
            if (kind.equals("SCORING"))
                rows.sort(
                        Comparator.<Map<String, Object>, java.math.BigDecimal>comparing(
                                        r -> new java.math.BigDecimal(r.get("score").toString()))
                                .reversed()
                                .thenComparing(r -> r.get("ruleId").toString()));
            if (rows.size() > max) rows = new ArrayList<>(rows.subList(0, max));
            steps.add(step("RULE_EXECUTION", "冻结规则计算完成，返回 " + rows.size() + " 行"));
        }
        return new Result(columns, rows, steps);
    }
}

package cn.ontorepo.trace.application;

import cn.ontorepo.common.domain.*;
import cn.ontorepo.contract.*;
import cn.ontorepo.data.domain.LabelScope;
import cn.ontorepo.operation.domain.FrozenPolicy;
import cn.ontorepo.space.application.GuardService;

import com.fasterxml.jackson.databind.*;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.util.*;

/** 供给编译与统一冻结守卫复核；不依赖任何草稿操作或客户端主体属性。 */
@Service
public class SupplyService {
    private static final ObjectMapper JSON = new ObjectMapper();
    private final GuardService guards;
    private final String baseUrl;

    public SupplyService(
            GuardService guards,
            @Value("${onto.service-base-url:http://127.0.0.1:18182}") String baseUrl) {
        this.guards = guards;
        this.baseUrl = baseUrl;
    }

    public JsonNode operation(Map<String, Object> p, Map<String, Object> op) {
        var node = JSON.createObjectNode();
        node.put("code", op.get("code").toString());
        node.set("input_schema", inputSchema(op));
        node.set("output_schema", JSON.valueToTree(op.get("outputSchema")));
        node.set("assistant_tool", JSON.valueToTree(op.get("assistantTool")));
        var policy = FrozenPolicy.of(p, op);
        node.set(
                "policy",
                JSON.valueToTree(
                        Map.of("max_rows", policy.maxRows(), "timeout_ms", policy.timeoutMs())));
        return node;
    }

    @SuppressWarnings("unchecked")
    private JsonNode cfg(Map<String, Object> p) {
        return JSON.valueToTree(
                ((Map<String, Object>) p.get("space")).getOrDefault("guardCfg", Map.of()));
    }

    /** MODEL_ONLY能力不能生成供给契约；冻结格式缺标记的旧候选也绝不默认通过。 */
    public List<JsonNode> manifests(Map<String, Object> p, String candidate) {
        if (!"CONTRACT".equals(p.get("capability")))
            throw new Problem(409, "CONTRACT_NOT_AVAILABLE", "该候选仅具备MODEL_ONLY本地模型能力");
        List<JsonNode> result = new ArrayList<>();
        for (var op : cn.ontorepo.knowledge.application.FrozenKnowledge.rows(p, "operations"))
            result.add(preview(p, op, candidate));
        return result;
    }

    private JsonNode preview(Map<String, Object> p, Map<String, Object> op, String candidate) {
        return new ContractGenerator().preview(operation(p, op), cfg(p), baseUrl, "", candidate);
    }

    /** 六步：入口会话、角色、冻结守卫完整性、当前主体授权、空间/策略范围、执行绑定适用性。 */
    public void authorize(
            UUID u,
            UUID s,
            Map<String, Object> p,
            Map<String, Object> op,
            String caller,
            String candidate) {
        var grant = guards.grant(u, s);
        JsonNode node = operation(p, op);
        boolean declared = op.get("assistantTool") != null;
        if ("SERVICE".equals(caller) && !"CONTRACT".equals(p.get("capability")))
            throw new Problem(409, "CONTRACT_NOT_AVAILABLE", "MODEL_ONLY候选不可作为内部供给服务调用");
        if (!declared) {
            if ("SERVICE".equals(caller)
                    || !Set.of("OBJECT_QUERY", "RELATION_QUERY", "RULE_CHECK", "SCORING")
                            .contains(op.getOrDefault("kind", "OBJECT_QUERY").toString()))
                throw new Problem(409, "ASSISTANT_REQUIRED", "外部操作必须具备完整助手声明与守卫");
            // 纯P1本地兼容仍不能绕过明确授权范围。
            if (!LabelScope.parse(JSON.valueToTree(grant.get("dataScope"))).clauses().isEmpty())
                throw new Problem(403, "GUARD_BINDING_NOT_APPLICABLE", "模型操作不能证明当前数据范围");
            return;
        }
        GuardValidator.validate(node, cfg(p));
        preview(p, op, candidate);
        JsonNode attrs = JSON.valueToTree(grant.get("attributes"));
        LabelScope scope = LabelScope.parse(JSON.valueToTree(grant.get("dataScope")));
        for (Object extra : FrozenPolicy.of(p, op).scopes())
            scope = LabelScope.intersect(scope, LabelScope.parse(JSON.valueToTree(extra)));
        Set<String> checked = new HashSet<>();
        for (JsonNode attr : node.path("assistant_tool").path("guard_attrs")) {
            String name = attr.asText();
            JsonNode def = cfg(p).path(name), rule = def.path("authorization_rule");
            // 即使同时存在主体授权规则，模型路径也不能静默丢掉声明的源列过滤。
            if (def.has("filter_binding")
                    || !"PRINCIPAL_ATTRIBUTE".equals(rule.path("kind").asText()))
                throw new Problem(403, "GUARD_BINDING_NOT_APPLICABLE", "模型查询不支持数据源列过滤守卫");
            JsonNode value = attrs.path(rule.path("attribute").asText());
            GuardService.typed(def.path("type").asText(), value);
            if (!scope.allows(name, value))
                throw new Problem(403, "DATA_SCOPE_DENIED", "当前主体属性超出授权范围");
            checked.add(name);
        }
        if (!checked.containsAll(scope.clauses().keySet()))
            throw new Problem(403, "GUARD_BINDING_NOT_APPLICABLE", "模型查询不能证明未绑定的数据范围");
    }

    public void input(Map<String, Object> op, Object value) {
        if (op.get("assistantTool") != null)
            JsonSchemaAssertions.validate(inputSchema(op), JSON.valueToTree(value));
    }

    /** 规则/评分只消费属性对象；在原Schema根追加合取约束，保留原类型和所有本地引用路径。 */
    private JsonNode inputSchema(Map<String, Object> op) {
        JsonNode schema = JSON.valueToTree(op.get("inputSchema"));
        if (Set.of("RULE_CHECK", "SCORING").contains(op.getOrDefault("kind", "OBJECT_QUERY"))
                && schema instanceof com.fasterxml.jackson.databind.node.ObjectNode object) {
            JsonSchemaAssertions.validateSchema(schema);
            object.withArray("allOf").addObject().put("type", "object");
        }
        return schema;
    }

    public void output(Map<String, Object> op, Object value) {
        if (op.get("assistantTool") != null)
            JsonSchemaAssertions.validate(
                    JSON.valueToTree(op.get("outputSchema")), JSON.valueToTree(value));
    }
}

    @Transactional
    public Map<String, Object> freeze(UUID user, UUID space, long rev) {
        spaces.lock(user, space, rev, true);
        Map<String, Object> payload = new TreeMap<>();
        payload.put(
                "space",
                db.one(
                        "SELECT"
                            + " id,code,name,namespace_uri,domain,status,draft_rev,created_at,guard_cfg"
                            + " FROM ontology_space WHERE id=?",
                        space));
        for (String[] table :
                new String[][] {
                    {"classes", "ont_class"},
                    {"properties", "ont_property"},
                    {"operations", "ont_operation"},
                    {"codebooks", "ont_codebook"},
                    {"relations", "ont_relation"},
                    {"constraints", "ont_constraint"},
                    {"rules", "ont_rule"},
                    {"strategies", "ont_strategy"},
                    {"testCases", "ont_test_case"},
                    {"rdfDocuments", "ont_rdf_document"},
                    {"knowledge", "ont_knowledge"}
                })
            payload.put(
                    table[0],
                    db.list("SELECT * FROM " + table[1] + " WHERE space_id=? ORDER BY id", space));
        // 新格式显式记录供给能力；不修改任何已经存在的候选原始字节。
        payload.put("formatVersion", 2);
        boolean contract =
                codec.rows(payload, "operations").stream()
                        .anyMatch(op -> op.get("assistantTool") != null);
        payload.put("capability", contract ? "CONTRACT" : "MODEL_ONLY");
        byte[] bytes = codec.encode(payload);
        UUID id = UUID.randomUUID();
        db.update(
                "INSERT INTO"
                    + " publish_candidate(id,space_id,draft_rev,sha256,payload,format_version,capability)"
                    + " VALUES(?,?,?,?,?,2,?)",
                id,
                space,
                rev,
                AuthService.digest(bytes),
                bytes,
                payload.get("capability"));
        spaces.audit(user, space, "CANDIDATE_FREEZE", rev);
        return db.one(
                "SELECT id,space_id,draft_rev,sha256,status,created_at,format_version,capability"
                        + " FROM publish_candidate WHERE id=?",
                id);

    /** 常量时间比较摘要；不以重新序列化结果校验，避免掩盖原始字节损坏。 */
    public Map<String, Object> payload(Map<String, Object> c) {
        byte[] bytes = (byte[]) c.get("payload");
        byte[] actual = AuthService.digest(bytes).getBytes(StandardCharsets.US_ASCII),
                expected = c.get("sha256").toString().getBytes(StandardCharsets.US_ASCII);
        if (!MessageDigest.isEqual(actual, expected))
            throw new Problem(409, "CANDIDATE_CORRUPT", "候选摘要不匹配，已拒绝试执行");
        return codec.decode(bytes);
    }

    public Map<String, Object> run(

    }

    /** SERVICE按操作码选择也必须处于同一事实边界，完整性失败时仍保留请求码与实际主体。 */
    private Map<String, Object> run(
            UUID user,
            UUID id,
            UUID operation,
            Object input,
            String caller,
            String question,
            String requestedCode) {
        var c = candidate(user, id);
        UUID space = Input.uuid(c.get("spaceId"));
        spaces.editor(user, space);
        var locked =
                db.one("SELECT status,draft_rev FROM ontology_space WHERE id=? FOR UPDATE", space);
        if (!locked.get("status").equals("ACTIVE"))
            throw new Problem(409, "SPACE_ARCHIVED", "归档空间不能试执行");
        UUID execution = UUID.randomUUID(), trace = UUID.randomUUID();
        long start = System.nanoTime();
        String status = "SUCCEEDED",
                error = null,
                opCode = requestedCode == null ? "UNKNOWN" : requestedCode;
        String frozenSpaceCode = null;
        List<String> columns = new ArrayList<>();
        List<Map<String, Object>> rows = new ArrayList<>();
        List<Map<String, String>> steps = new ArrayList<>();
        RuntimeException rejection = null;
        try {
            var p = payload(c);
            frozenSpaceCode = Objects.toString(((Map<?, ?>) p.get("space")).get("code"), "");
            steps.add(step("VERIFY_PAYLOAD", "SUCCEEDED", "冻结载荷摘要复核通过"));
            if (question != null) {
                try {
                    operation = matcher.select(p, question);
                    steps.add(step("TERM_MATCH", "SUCCEEDED", "冻结术语匹配操作 " + operation));
                } catch (Problem e) {
                    steps.add(step("TERM_MATCH", "FAILED", e.getMessage()));
                    throw e;
                }
            }
            final UUID selectedOperation = operation;
            var op =
                    codec.rows(p, "operations").stream()
                            .filter(
                                    o ->
                                            requestedCode != null
                                                    ? requestedCode.equals(o.get("code"))
                                                    : selectedOperation != null
                                                            && o.get("id")
                                                                    .equals(
                                                                            selectedOperation
                                                                                    .toString()))
                            .findFirst()
                            .orElseThrow(
                                    () ->
                                            new Problem(
                                                    "SERVICE".equals(caller) ? 404 : 400,
                                                    "OPERATION_NOT_IN_CANDIDATE",
                                                    "操作不在当前候选中"));
            opCode = op.get("code").toString();
            operation = Input.uuid(op.get("id"));
            // 仅旧的无Schema/无助手操作沿用P1显式null视为空属性对象；声明nullable输入保留真实null。
            if (input == null && op.get("inputSchema") == null && op.get("assistantTool") == null)
                input = Map.of();
            supply.authorize(user, space, p, op, caller, id.toString());
            // MODEL_ONLY保留P1三步骤事实格式；实际授权仍必经上方统一检查。
            if (op.get("assistantTool") != null)
                steps.add(step("AUTHORIZE_GUARDS", "SUCCEEDED", "冻结守卫与当前空间授权复核通过"));
            supply.input(op, input);
            var evaluated = executor.execute(p, op, input);
            columns.addAll(evaluated.columns());
            for (var row : evaluated.rows()) rows.add(new java.util.LinkedHashMap<>(row));
            steps.addAll(evaluated.steps());
            supply.output(op, Map.of("columns", columns, "rows", rows));
        } catch (ContractProblem e) {
            status = "FAILED";
            error = e.code();
            rejection = e;
            columns.clear();
            rows.clear();
            steps.add(step("CONTRACT_VALIDATE", "FAILED", e.getMessage()));
        } catch (Problem e) {
            if (e.code.equals("OPERATION_NOT_IN_CANDIDATE") && !"SERVICE".equals(caller)) throw e;
            status = "FAILED";
            error = e.code;
            columns.clear();
            rows.clear();
            steps.add(step("EXECUTE", "FAILED", e.getMessage()));
            if ("SERVICE".equals(caller)
                    || e.code.startsWith("GUARD_")
                    || e.code.equals("DATA_SCOPE_DENIED")
                    || e.code.equals("CONTRACT_NOT_AVAILABLE")
                    || e.code.equals("ASSISTANT_REQUIRED")) rejection = e;
        } catch (QueryCancelledException e) {
            status = "FAILED";
            error = "QUERY_TIMEOUT";
            columns.clear();
            rows.clear();
            steps.add(step("MODEL_SELECT", "FAILED", "查询超过执行时限"));
        } catch (Exception e) {
            status = "FAILED";
            error = "EXECUTION_FAILED";
            columns.clear();
            rows.clear();
            steps.add(step("MODEL_SELECT", "FAILED", "模型查询失败"));
        }
        long elapsed = TimeUnit.NANOSECONDS.toMillis(System.nanoTime() - start);
        Map<String, Object> result = Map.of("columns", columns, "rows", rows);
        // 输入摘要包含冻结主体、操作标识及空参数，输出摘要包含状态、错误与实际结果。
        // 不把当前草稿修订加入输入，确保相同候选试执行事实可独立复核。
        Map<String, Object> inputFact = new TreeMap<>();
        inputFact.put("candidateId", id.toString());
        inputFact.put("candidateSha256", c.get("sha256"));
        inputFact.put("operationId", operation == null ? "UNRESOLVED" : operation.toString());
        inputFact.put("parameters", input);
        if (requestedCode != null) inputFact.put("operationCode", requestedCode);
        if (question != null) inputFact.put("question", question);
        String inputDigest = AuthService.digest(codec.encode(inputFact));
        Map<String, Object> outputFact = new TreeMap<>();
        outputFact.put("status", status);
        outputFact.put("errorCode", error);
        outputFact.put("result", result);
        String outputDigest = AuthService.digest(codec.encode(outputFact));
        db.update(
                "INSERT INTO"
                    + " op_execution(id,space_id,trace_id,subject_type,subject_id,op_code,status,error_code,elapsed_ms,candidate_sha256,result_json,principal_id,caller,input_digest,output_digest)"
                    + " VALUES(?,?,?,'CANDIDATE',?,?,?,?,?,?,?::jsonb,?,?,?,?)",
                execution,
                space,
                trace,
                id,
                opCode,
                status,
                error,
                elapsed,
                c.get("sha256"),
                codec.json(result),
                user,
                caller,
                inputDigest,
                outputDigest);
        int order = 0;
        for (var step : steps)
            db.update(
                    "INSERT INTO op_execution_step(execution_id,step_order,kind,status,detail)"
                            + " VALUES(?,?,?,?,?)",
                    execution,
                    order++,
                    step.get("kind"),
                    step.get("status"),
                    step.get("detail"));
        spaces.audit(user, space, "CANDIDATE_TRIAL", ((Number) locked.get("draftRev")).longValue());
        // 外部事务声明noRollbackFor，拒绝事实提交后HTTP边界才能抛出原始结构化错误。
        if (rejection != null && !"LOCAL_TEST".equals(caller)) throw rejection;
        var fact = execution(execution);
        fact.put("spaceCode", frozenSpaceCode);
        return fact;
    }

    /** 内部入口按冻结opCode选择；业务data不包含执行id，meta来自真实候选与空间上下文。 */
    @Transactional(noRollbackFor = {Problem.class, ContractProblem.class})
    public Map<String, Object> service(UUID user, UUID id, String opCode, Object input) {
        var fact = run(user, id, null, input, "SERVICE", null, opCode);
        if (!"SUCCEEDED".equals(fact.get("status")))
            throw new Problem(
                    400, Objects.toString(fact.get("errorCode"), "EXECUTION_FAILED"), "冻结操作执行失败");
        Map<String, Object> meta = new java.util.LinkedHashMap<>();
        meta.put("spaceCode", fact.get("spaceCode"));
        meta.put("releaseId", null);
        meta.put("version", null);
        meta.put("contentDigest", fact.get("candidateSha256"));
        meta.put("preview", true);
        return Map.of(
                "code",
                "OK",
                "data",
                Map.of("columns", fact.get("columns"), "rows", fact.get("rows")),
                "meta",
                meta);
    }

    private Map<String, String> step(String kind, String status, String detail) {
        return Map.of("kind", kind, "status", status, "detail", detail);
    }

    /** 执行历史按空间授权读取，首批仅保留最近50条返回，不暴露内部身份事实字段。 */
    public List<Map<String, Object>> executions(UUID user, UUID space) {
        spaces.role(user, space);
        return db
                .list(
                        "SELECT id FROM op_execution WHERE space_id=? ORDER BY created_at DESC"
                                + " LIMIT 50",
                        space)
                .stream()
                .map(r -> execution(Input.uuid(r.get("id"))))
                .toList();
    }

    private Map<String, Object> execution(UUID id) {
        var r =
                db.one(
                        "SELECT"
                            + " id,trace_id,subject_type,subject_id,op_code,status,error_code,elapsed_ms,created_at,candidate_sha256,result_json::text"
                            + " FROM op_execution WHERE id=?",
                        id);
        r.putAll(codec.decode(r.remove("resultJson").toString().getBytes(StandardCharsets.UTF_8)));
        r.put(
                "steps",
                db.list(
                        "SELECT kind,status,detail FROM op_execution_step WHERE execution_id=?"
                                + " ORDER BY step_order",
                        id));
        return r;
    }
}

/** 用例只使用候选冻结的 enabled 集合，确定性比较结构并持久化 LOCAL_TEST 执行事实。 */
@Service
public class CandidateTestService {
    private final CandidateService candidates;
    private final SpaceService spaces;
    private final Db db;

        this.candidates = candidates;

    @SuppressWarnings("unchecked")
    @Transactional
    public Map<String, Object> tests(UUID user, UUID id) {
        var candidate = candidates.candidate(user, id);
        UUID space = Input.uuid(candidate.get("spaceId"));
        spaces.editor(user, space);
        Map<String, Object> payload;
        try {
            payload = candidates.payload(candidate);
        } catch (Problem e) {
            if (!e.code.equals("CANDIDATE_CORRUPT")) throw e;
            // 不能从损坏载荷读取或伪造用例；登记一次LOCAL_TEST完整性失败并关联实际执行事实。
            var failed = candidates.run(user, id, null, Map.of(), "LOCAL_TEST");
            return Map.of(
                    "candidateSha256",
                    candidate.get("sha256"),
                    "passed",
                    false,
                    "results",
                    List.of(
                            Map.of(
                                    "caseId",
                                    "",
                                    "name",
                                    "候选完整性检查",
                                    "passed",
                                    false,
                                    "executionId",
                                    failed.get("id"),
                                    "message",
                                    e.getMessage())));
        }
        List<Map<String, Object>> results = new ArrayList<>();
        boolean all = true;
        for (var test : RdfCodec.rows(payload, "testCases")) {
            if (!Boolean.TRUE.equals(test.get("enabled"))) continue;
            String executionId = "";
            boolean passed = false;
            String message = "";
            try {
                var execution =
                        candidates.run(
                                user,
                                id,
                                "NL".equals(test.get("mode"))
                                        ? null
                                        : Input.uuid(test.get("operationId")),
                                test.get("input"),
                                "LOCAL_TEST",
                                "NL".equals(test.get("mode"))
                                        ? test.get("question").toString()
                                        : null);
                executionId = execution.get("id").toString();
                Map<String, Object> expected = (Map<String, Object>) test.get("expected");
                Map<String, Object> actual = new LinkedHashMap<>();
                for (String key : List.of("status", "errorCode", "opCode", "columns", "rows"))
                    if (expected.containsKey(key)) actual.put(key, execution.get(key));
                boolean allowed =
                        Set.of("status", "errorCode", "opCode", "columns", "rows")
                                .containsAll(expected.keySet());
                passed =
                        allowed
                                && !expected.isEmpty()
                                && RdfCodec.rows(payload, "operations").stream()
                                        .anyMatch(
                                                op ->
                                                        Input.sameUuid(
                                                                        op.get("id"),
                                                                        test.get("operationId"))
                                                                && Objects.equals(
                                                                        op.get("code"),
                                                                        execution.get("opCode")))
                                && Objects.equals(
                                        expected.getOrDefault("status", "SUCCEEDED"),
                                        execution.get("status"))
                                && ("FAILED".equals(expected.get("status"))
                                        || "SUCCEEDED".equals(execution.get("status")))
                                && matches(
                                        expected, actual, "CONTAINS".equals(test.get("matchMode")));
                message = passed ? "操作命中且结果断言匹配" : "操作结果不符合冻结断言";
            } catch (ContractProblem e) {
                message = e.getMessage();
            } catch (Problem e) {
                message = e.getMessage();
            }
            all &= passed;
            results.add(
                    Map.of(
                            "caseId",
                            test.get("id"),
                            "name",
                            test.get("name"),
                            "passed",
                            passed,
                            "executionId",

                            "message",

        }

    }

    /** EXACT 逐层相等；CONTAINS 对象按字段子集、数组按元素包含，数值按数学值比较。 */
    private boolean matches(Object expected, Object actual, boolean contains) {
        if (expected instanceof Map<?, ?> em) {
            if (!(actual instanceof Map<?, ?> am) || !contains && em.size() != am.size())
                return false;
            for (var e : em.entrySet())
                if (!am.containsKey(e.getKey())
                        || !matches(e.getValue(), am.get(e.getKey()), contains)) return false;
            return true;
        }
        if (expected instanceof List<?> el) {
            if (!(actual instanceof List<?> al)) return false;
            if (contains) {
                List<Object> left = new ArrayList<>(al);
                for (Object e : el) {
                    int found = -1;
                    for (int i = 0; i < left.size(); i++)
                        if (matches(e, left.get(i), true)) {
                            found = i;
                            break;
                        }
                    if (found < 0) return false;
                    left.remove(found);
                }
                return true;
            }
            if (el.size() != al.size()) return false;
            for (int i = 0; i < el.size(); i++)
                if (!matches(el.get(i), al.get(i), false)) return false;
            return true;
        }
        return PropertyFacets.equal(expected, actual);
    }
}
