import { useSession } from '../../stores/session'
import { useWorkspace } from '../../stores/workspace'
import { useResource } from '../../composables/useResource'
import { useDraftRevision } from '../../composables/useDraftRevision'
import { readDraftSnapshot } from '../../composables/readDraftSnapshot'
import { useMutation, notifySaved } from '../../composables/useMutation'
import type { Space } from '@onto/api-client'

/** 单文档编辑的修订基线来自完整一致性读取；409 后不自动重读或替换用户输入。 */
export function useDraftDocument<T>(read: (spaceId: string) => Promise<T>) {
  const api = useSession().api,
    workspace = useWorkspace()
  const resource = useResource<{ items: T; space: Space } | null>(null)
  const draft = useDraftRevision(),
    mutation = useMutation()
  async function load() {
    const id = workspace.space?.id
    if (!id) return
    const result = await resource.load(() => readDraftSnapshot(api, id, () => read(id)))
    if (result) {
      draft.start(result.space.draftRev)
      workspace.refreshed(result.space)
      mutation.error.value = ''
    }
    return result
  }
  async function save(write: (expectedDraftRev: number) => Promise<Space>) {
    if (!resource.data.value || !workspace.canWrite || draft.stale.value) return false
    const ok = await mutation.run(() =>
      draft.submit(async (rev) => workspace.accept(await write(rev))),
    )
    if (ok) notifySaved()
    return ok
  }
  return { api, workspace, resource, draft, mutation, load, save }
}

/** 独立内部调用入口：body 只有业务输入，调用身份和可信属性始终由服务器取得。 */
import { ref, watch } from 'vue'
import { ApiError, type Candidate, type Operation, type Envelope } from '@onto/api-client'
import { useSession } from '../../stores/session'
import { useWorkspace } from '../../stores/workspace'
import { useResource } from '../../composables/useResource'
import { useMutation } from '../../composables/useMutation'
import { objectField, FieldError } from './fields'
import { verifyInvocationMeta } from './candidateIdentity'
import AsyncState from '@onto/shared-ui/AsyncState.vue'
const props = defineProps<{ candidate: Candidate }>(),
  api = useSession().api,
  workspace = useWorkspace()
const operations = useResource<Operation[]>([]),
  mutation = useMutation()
const code = ref(''),
  input = ref('{}'),
  inputError = ref(''),
  result = ref<Envelope<unknown> | null>(null)
let sequence = 0
function loadOperations() {
  const id = props.candidate.id
  return operations.load(() => api.get<Operation[]>(`/candidates/${id}/operations`))
}
watch(
  () => `${props.candidate.id}:${props.candidate.sha256}`,
  () => {
    sequence++
    operations.reset()
    code.value = ''
    input.value = '{}'
    result.value = null
    mutation.error.value = ''
    inputError.value = ''
    void loadOperations()
  },
  { immediate: true },
)
async function execute() {
  if (
    !workspace.canWrite ||
    !operations.data.value.some((operation) => operation.code === code.value)
  )
    return
  const { id, sha256 } = props.candidate,
    ticket = sequence,
    opCode = code.value
  result.value = null
  inputError.value = ''
  await mutation.run(async () => {
    try {
      const value = objectField(input.value, 'input', '操作输入')
      const response = await api.requestEnvelope<unknown>(
        `/internal/candidates/${id}/ops/${encodeURIComponent(opCode)}`,
        'POST',
        value,
      )
      verifyInvocationMeta(response.meta, sha256)
      if (ticket === sequence) result.value = response
    } catch (cause) {
      if (ticket !== sequence) return
      if (cause instanceof FieldError) inputError.value = cause.message
      if (cause instanceof ApiError) throw new Error(`${cause.code}：${cause.message}`)
      throw cause
    }
  })
}

import type { JsonObject } from '@onto/api-client'
import { parseObject } from '../p1/inputs'

/** 用户编辑错误只定位到具体字段，原始文本继续保留供修复或复制。 */
export class FieldError extends Error {
  constructor(
    public readonly field: string,
    message: string,
  ) {
    super(message)
  }
}
export function objectField(text: string, field: string, label: string): JsonObject {
  try {
    return parseObject(text, label)
  } catch (cause) {
    throw new FieldError(field, cause instanceof Error ? cause.message : `${label}格式错误。`)
  }
}
export function objectArrayField(text: string, field: string, label: string): JsonObject[] {
  let value: unknown
  try {
    value = JSON.parse(text)
  } catch {
    throw new FieldError(field, `${label}须为有效的 JSON 数组。`)
  }
  if (
    !Array.isArray(value) ||
    value.some((item) => !item || typeof item !== 'object' || Array.isArray(item))
  )
    throw new FieldError(field, `${label}须为 JSON 数组，每项为对象。`)
  return value as JsonObject[]
}
export function lines(text: string) {
  return [
    ...new Set(
      text
        .split(/\r?\n/)
        .map((item) => item.trim())
        .filter(Boolean),
    ),
  ]
}

/** 所有 HTTP、信封与网络失败共用此错误，页面据此提供刷新或重新登录。 */
export class ApiError extends Error {
  constructor(
    public readonly code: string,
    message: string,
    public readonly status: number,
  ) {
    super(message)
    this.name = 'ApiError'
  }
}
export interface Envelope<T> {
  code: string
  message?: string
  data: T
  meta?: unknown
}
export class ApiClient {
  constructor(
    private readonly token: () => string | null,
    private readonly expired: () => void,
    private readonly fetcher: typeof fetch = fetch,
  ) {}
  /** 令牌仅在请求时读取内存；不接触 localStorage/sessionStorage。 */
  async request<T>(path: string, method = 'GET', body?: unknown): Promise<T> {
    return (await this.requestEnvelope<T>(path, method, body)).data
  }
  /** 内部候选调用需保留 preview 与摘要元数据，业务结果仍独立置于 data。 */
  async requestEnvelope<T>(path: string, method = 'GET', body?: unknown): Promise<Envelope<T>> {
    const headers = new Headers({ Accept: 'application/json' })
    const token = this.token()
    if (token) headers.set('Authorization', `Bearer ${token}`)
    if (body !== undefined) headers.set('Content-Type', 'application/json')
    let response: Response
    try {
      // 原生 fetch 按全局函数调用，避免实例代理改变浏览器要求的调用上下文。
      const fetcher = this.fetcher
      response = await fetcher(`/onto${path}`, {
        method,
        headers,
        body: body === undefined ? undefined : JSON.stringify(body),
      })
    } catch {
      throw new ApiError('NETWORK_ERROR', '无法连接服务，请检查网络后重试。', 0)
    }
    // 旧会话的迟到响应只能使旧请求失败，不能注销用户重新登录后的新会话。
    if (response.status === 401 && token === this.token()) this.expired()
    let envelope: Envelope<T>
    try {
      const parsed: unknown = await response.json()
      if (
        typeof parsed !== 'object' ||
        parsed === null ||
        !('code' in parsed) ||
        typeof parsed.code !== 'string' ||
        ('message' in parsed && typeof parsed.message !== 'string') ||
        (parsed.code !== 'OK' && !('message' in parsed)) ||
        !('data' in parsed)
      )
        throw new Error('Invalid envelope')
      envelope = parsed as Envelope<T>
    } catch {
      throw new ApiError('INVALID_RESPONSE', '服务响应无法读取，请稍后重试。', response.status)
    }
    if (!response.ok || envelope.code !== 'OK')
      throw new ApiError(
        envelope.code || 'HTTP_ERROR',
        envelope.message ||
          (response.status === 403 ? '当前账号没有此操作权限。' : '操作失败，请重试。'),
        response.status,
      )
    return envelope
  }
  get<T>(path: string) {
    return this.request<T>(path)
  }
  post<T>(path: string, body: unknown) {
    return this.request<T>(path, 'POST', body)
  }
  put<T>(path: string, body: unknown) {
    return this.request<T>(path, 'PUT', body)
  }
  delete<T>(path: string) {
    return this.request<T>(path, 'DELETE')
  }
}

  mode: 'DIRECT' as NonNullable<TestCase['mode']>,
})

function edit(item: TestCase | null) {
  page.edit(item)
  Object.assign(form, {
    name: item?.name || '',
    question: item?.question || '',
    operationId: item?.operationId || '',
    input: JSON.stringify(item?.input || {}, null, 2),
    expected: JSON.stringify(item?.expected || {}, null, 2),
    matchMode: item?.matchMode || 'EXACT',
    enabled: item?.enabled ?? true,
    mode: item?.mode || 'DIRECT',
  })
}
async function save() {
  try {
    if (!form.name.trim() || !form.question.trim() || !form.operationId)
      throw new Error('请填写名称、问题并选择预期命中操作。')

      ...form,
      input: parseObject(form.input, '用例输入'),
      expected: parseObject(form.expected, '预期结果'),
    })
  } catch (cause) {
    page.mutation.error.value = cause instanceof Error ? cause.message : '用例无法读取。'
  }
}
