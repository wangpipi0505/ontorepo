import { useResource } from '../../composables/useResource'
import { readDraftSnapshot } from '../../composables/readDraftSnapshot'
import { useSession } from '../../stores/session'
import { useWorkspace } from '../../stores/workspace'
interface NamedReference {
  id: string
  name: string
}
/** 选择器只读服务端同空间条目；失败直接展示，不提供本地演示条目，也不解锁旧表单。 */
export function useReferences(paths: string[]) {
  const resource = useResource<Record<string, NamedReference[]>>({})
  async function load() {
    const id = useWorkspace().space?.id,
      api = useSession().api
    if (!id) return
    return resource.load(async () => {
      const result = await readDraftSnapshot(api, id, async () => {
        // 原属性列表要求 classId；全空间选择器必须使用独立只读入口，保留旧接口400语义。
        const pairs = await Promise.all(
          paths.map(
            async (path) =>
              [
                path,
                await api.get<NamedReference[]>(
                  path === 'properties' ? `/spaces/${id}/properties` : `/${path}?spaceId=${id}`,
                ),
              ] as const,
          ),
        )
        return Object.fromEntries(pairs)
      })
      return result.items
    })
  }
  return { ...resource, load }
}

import { ref, shallowRef } from 'vue'
import type { Space } from '@onto/api-client'
import { useWorkspace } from '../../stores/workspace'
import { useSession } from '../../stores/session'
import { useResource } from '../../composables/useResource'
import { useDraftRevision } from '../../composables/useDraftRevision'
import { useMutation, confirmAction, notifySaved } from '../../composables/useMutation'
import { readDraftSnapshot } from '../../composables/readDraftSnapshot'

/** 仅复用列表的并发/提交不变量；领域字段仍由独立业务表单维护，避免通用巨型表单。 */
export function useDraftCollection<T extends { id: string; name: string }>(path: string) {
  const workspace = useWorkspace(),
    api = useSession().api
  const items = useResource<T[]>([]),
    mutation = useMutation(),
    draft = useDraftRevision()
  const open = ref(false),
    editing = shallowRef<T | null>(null)
  let loadedRev: number | undefined
  async function load() {
    const spaceId = workspace.space?.id
    if (!spaceId) return
    let current: Space | undefined
    const result = await items.load(async () => {
      const snapshot = await readDraftSnapshot(api, spaceId, () =>
        api.get<T[]>(`${path}?spaceId=${spaceId}`),
      )
      current = snapshot.space
      return snapshot.items
    })
    if (result && current) {
      loadedRev = current.draftRev
      workspace.refreshed(current)
    }
  }
  function edit(item: T | null) {
    if (!workspace.canWrite) return
    editing.value = item
    draft.start(loadedRev ?? workspace.space?.draftRev ?? 0)
    mutation.error.value = ''
    open.value = true
  }
  async function save(body: Record<string, unknown>) {
    const space = workspace.space
    if (!space || !workspace.canWrite) return false
    const ok = await mutation.run(() =>
      draft.submit(async (expectedDraftRev) => {
        const request = { ...body, spaceId: space.id, expectedDraftRev }
        const next = editing.value
          ? await api.put<Space>(`${path}/${editing.value.id}`, request)
          : await api.post<Space>(path, request)
        workspace.accept(next)
      }),
    )
    if (ok) {
      open.value = false
      notifySaved()
      await load()
    }
    return ok
  }
  async function remove(item: T) {
    const space = workspace.space
    if (!space || !workspace.canWrite || loadedRev === undefined) return
    if (!(await confirmAction(`删除“${item.name}”？受引用条目可能被服务端阻止。`))) return
    await mutation.run(async () => {
      workspace.accept(
        await api.delete<Space>(
          `${path}/${item.id}?spaceId=${space.id}&expectedDraftRev=${loadedRev}`,
        ),
      )
      await load()
    })
  }
  return { workspace, api, items, mutation, draft, open, editing, load, edit, save, remove }
}

import type { RangeType, PropertyFacets, Scalar } from '@onto/api-client'
/** JSON 编辑器不执行表达式；非法输入保留原文并阻止提交，不以空对象兜底。 */
export function parseObject(text: string, label: string): Record<string, unknown> {
  try {
    const value: unknown = JSON.parse(text)
    if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error()
    return value as Record<string, unknown>
  } catch {
    throw new Error(`${label}须为有效的 JSON 对象。`)
  }
}
/** 保留 JSON 标量类型。服务端仍负责码表同空间、引用、单类唯一主键等权威校验。 */
export function parseFacets(type: RangeType, text: string, required: boolean): PropertyFacets {
  const value = parseObject(text, '高级分面')
  const valid = (item: unknown): item is Scalar => {
    if (type === 'INTEGER') return typeof item === 'number' && Number.isInteger(item)
    if (type === 'DECIMAL') return typeof item === 'number' && Number.isFinite(item)
    if (type === 'BOOLEAN') return typeof item === 'boolean'
    return typeof item === 'string' && (type !== 'DATE' || /^\d{4}-\d{2}-\d{2}$/.test(item))
  }
  if ('defaultValue' in value && !valid(value.defaultValue))
    throw new Error('默认值与属性类型不匹配。')
  if ('enumValues' in value && (!Array.isArray(value.enumValues) || !value.enumValues.every(valid)))
    throw new Error('枚举值与属性类型不匹配。')
  for (const key of ['minimum', 'maximum'])
    if (key in value && (typeof value[key] !== 'number' || !['INTEGER', 'DECIMAL'].includes(type)))
      throw new Error('数值范围仅用于数值类型。')
  if (
    typeof value.minimum === 'number' &&
    typeof value.maximum === 'number' &&
    value.minimum > value.maximum
  )
    throw new Error('最小范围不能大于最大范围。')
  if (value.primaryKey === true && !required) throw new Error('主键必须设为必填。')
  if (
    'defaultValue' in value &&
    Array.isArray(value.enumValues) &&
    !value.enumValues.includes(value.defaultValue)
  )
    throw new Error('默认值必须属于枚举。')
  if (
    typeof value.defaultValue === 'number' &&
    ((typeof value.minimum === 'number' && value.defaultValue < value.minimum) ||
      (typeof value.maximum === 'number' && value.defaultValue > value.maximum))
  )
    throw new Error('默认值超出范围。')
  return value as PropertyFacets
}
export function parseEntries(text: string) {
  const seen = new Set<string>()
  const entries = text
    .split(/\r?\n/)
    .filter((line) => line.trim())
    .map((line) => {
      const index = line.indexOf('=')
      const code = line.slice(0, index).trim(),
        label = line.slice(index + 1).trim()
      if (index <= 0 || !code || !label) throw new Error('每行填写 编码=名称，二者不可为空。')
      if (seen.has(code)) throw new Error(`编码重复：${code}`)
      seen.add(code)
      return { code, label }
    })
  if (!entries.length) throw new Error('至少填写一个码表条目。')
  return entries
}

              props.multiple
