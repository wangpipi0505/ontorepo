/** 图与可读列表共享服务端事实；缺失边端点不补造节点，仍在关系列表展示。 */
import { computed, useId } from 'vue'
import type { TraceGraphData } from '../../../../../packages/api-client/src/release'
const props = defineProps<{ graph: TraceGraphData }>()
const marker = `arrow-${useId().replaceAll(':', '')}`
const nodes = computed(() =>
  props.graph.nodes.map((node, index) => ({
    ...node,
    x: index % 2 === 0 ? 25 : 345,
    y: 25 + Math.floor(index / 2) * 115,
  })),
)
const edges = computed(() =>
  props.graph.edges.flatMap((edge) => {
    const from = nodes.value.find((node) => node.id === edge.from),
      to = nodes.value.find((node) => node.id === edge.to)
    return from && to
      ? [{ ...edge, x1: from.x + 115, y1: from.y + 58, x2: to.x + 115, y2: to.y }]
      : []
  }),
)
const height = computed(() => Math.max(130, Math.ceil(nodes.value.length / 2) * 115 + 25))

<script setup lang="ts">
/** 正式调用使用 TOKEN 的实际主体，成功证据必须来自服务返回且匹配当前正式版本。 */
import { computed, ref, shallowRef, watch } from 'vue'
import type { Envelope } from '@onto/api-client'
import type { Endpoint, Release } from '../../../../../packages/api-client/src/release'
import { useSession } from '../../stores/session'
import { useWorkspace } from '../../stores/workspace'
import { useMutation } from '../../composables/useMutation'
import ParameterFields from './ParameterFields.vue'
import { parameterObject, type ParameterRow } from './parameters'
const props = defineProps<{ release: Release; endpoints: Endpoint[] }>()
const api = useSession().api,
  workspace = useWorkspace(),
  mutation = useMutation()
const code = ref(''),
  rows = ref<ParameterRow[]>([]),
  result = shallowRef<Envelope<unknown> | null>(null)
const inputMode = ref<'fields' | 'json'>('fields'),
  jsonInput = ref('{}')
const endpoint = computed(() => props.endpoints.find((item) => item.opCode === code.value))
const allowed = computed(
  () =>
    props.release.status === 'READY' &&
    !!workspace.space &&
    !!endpoint.value &&
    endpoint.value.status === 'ACTIVE',
)
let generation = 0
watch(
  () => props.release.id,
  () => {
    generation++
    code.value = ''
    rows.value = []
    result.value = null
    mutation.error.value = ''
  },
)
watch(code, () => {
  generation++
  rows.value = []
  jsonInput.value = '{}'
  inputMode.value = 'fields'
  result.value = null
  mutation.error.value = ''
})
function readInput(): unknown {
  if (inputMode.value === 'fields') return parameterObject(rows.value)
  try {
    return JSON.parse(jsonInput.value) as unknown
  } catch {
    throw new Error('操作输入须为有效 JSON 值。请检查格式后重试。')
  }
}
async function execute() {
  if (!allowed.value) return
  const release = props.release,
    opCode = code.value,
    ticket = generation,
    spaceCode = workspace.space?.code
  result.value = null
  await mutation.run(async ({ isCurrent }) => {
    const response = await api.requestEnvelope<unknown>(
      `/internal/releases/${release.id}/ops/${encodeURIComponent(opCode)}`,
      'POST',
      readInput(),
    )
    if (!isCurrent() || ticket !== generation) return
    const meta = response.meta
    if (
      !meta ||
      typeof meta !== 'object' ||
      !('preview' in meta) ||
      meta.preview !== false ||
      !('releaseId' in meta) ||
      meta.releaseId !== release.id ||
      !('version' in meta) ||
      meta.version !== `v${release.versionNo}` ||
      !('contentDigest' in meta) ||
      meta.contentDigest !== release.bundleSha256 ||
      !('spaceCode' in meta) ||
      meta.spaceCode !== spaceCode
    )
      throw new Error('服务返回的正式版本证据不完整或与所选版本不符，请刷新版本后重试。')
    result.value = response
  })
}
</script>
<template>
  <section class="p4-page panel stack">
    <h2>正式服务调用</h2>
    <p class="muted">
      以当前登录用户调用指定版本。OWNER 身份仍需有效的端点授权与空间授权；所有可信守卫由服务端取得。
    </p>
    <p v-if="release.status !== 'READY'" class="readonly-note">
      {{ release.status }} 版本不可消费。
    </p>
    <label>
      正式操作
      <select
        v-model="code"
        data-testid="service-operation"
        :disabled="mutation.pending.value || release.status !== 'READY'"
      >
        <option value="">请选择正式操作</option>
        <option v-for="item in endpoints" :key="item.id" :value="item.opCode">
          {{ item.opCode }} · {{ item.status }}
        </option>
      </select>
    </label>
    <p v-if="endpoint" class="identity">固定版本路由：{{ endpoint.route }}</p>
    <label>
      操作输入方式
      <select
        v-model="inputMode"
        data-testid="service-input-mode"
        :disabled="mutation.pending.value || !allowed"
      >
        <option value="fields">逐字段填写业务对象</option>
        <option value="json">直接输入 JSON 值（支持数组、标量和 null）</option>
      </select>
    </label>
    <ParameterFields
      v-if="inputMode === 'fields'"
      v-model="rows"
      :disabled="mutation.pending.value || !allowed"
      add-label="添加业务参数"
    />
    <label v-else>
      操作输入（JSON 值）
      <textarea
        v-model="jsonInput"
        data-testid="service-json-input"
        rows="5"
        :disabled="mutation.pending.value || !allowed"
      />
    </label>
    <div>
      <el-button
        type="primary"
        :disabled="!allowed"
        :loading="mutation.pending.value"
        @click="execute"
      >
        调用正式服务
      </el-button>
    </div>
    <p v-if="mutation.error.value" class="error-text" role="alert">{{ mutation.error.value }}</p>
    <div v-if="result" data-testid="service-result" class="stack">
      <h3>业务数据</h3>
      <pre>{{ JSON.stringify(result.data, null, 2) }}</pre>
      <h3>正式版本元数据</h3>
      <pre>{{ JSON.stringify(result.meta, null, 2) }}</pre>
    </div>
  </section>
</template>


## web/apps/onto-admin/src/features/p4\RestoreDraftPanel.vue
<script setup lang="ts">
/** 确认面板捕获打开时修订，409 或外部刷新后不能借新修订覆盖用户草稿。 */
import { computed, ref } from 'vue'
import { ApiError, type Space } from '@onto/api-client'
import type { Release } from '../../../../../packages/api-client/src/release'
import { useWorkspace } from '../../stores/workspace'
import { useSession } from '../../stores/session'
import { useMutation } from '../../composables/useMutation'
import './p4.css'
const props = defineProps<{ release: Release }>()
const emit = defineEmits<{ restored: []; cancel: [] }>()
const workspace = useWorkspace(),
  api = useSession().api,
  mutation = useMutation()
const expectedDraftRev = workspace.space?.draftRev,
  releaseId = props.release.id
const reason = ref(''),
  confirmed = ref(false),
  locked = ref(false),
  done = ref(false)
const allowed = computed(
  () =>
    workspace.space?.role === 'OWNER' &&
    workspace.canWrite &&
    !locked.value &&
    !done.value &&
    props.release.status !== 'CORRUPTED' &&
    expectedDraftRev === workspace.space?.draftRev &&
    releaseId === props.release.id,
)
async function restore() {
  if (!allowed.value || !confirmed.value || !reason.value.trim()) return
  await mutation.run(async ({ isCurrent }) => {
    try {
      const updated = await api.post<Space>(`/releases/${releaseId}/restore-draft`, {
        expectedDraftRev,
        reason: reason.value.trim(),
      })
      if (!isCurrent()) return
      workspace.accept(updated)
      done.value = true
      emit('restored')
    } catch (cause) {
      if (isCurrent() && cause instanceof ApiError && cause.status === 409) locked.value = true
      throw cause
    }
  })
}
</script>
<template>
  <section class="p4-page panel stack">
    <h2>恢复历史版本到草稿</h2>
    <p class="notice">
      目标 v{{ release.versionNo }} · {{ release.id }}。将覆盖打开面板时的草稿 r{{
        expectedDraftRev
      }}，生成新修订；历史版本保持不变，恢复后需要重新冻结和发布。
    </p>
    <p class="identity">
      目标摘要 {{ release.bundleSha256 }}
      <br />
      当前草稿 r{{ workspace.space?.draftRev }}
    </p>
    <label>
      恢复理由
      <textarea
        v-model="reason"
        data-testid="restore-reason"
        rows="3"
        :disabled="mutation.pending.value || done"
      />
    </label>
    <label class="check-label">
      <input
        v-model="confirmed"
        data-testid="restore-confirm"
        type="checkbox"
        :disabled="!allowed || mutation.pending.value"
      />
      确认使用此版本全量覆盖当前草稿内容
    </label>
    <p v-if="!allowed && !done" class="error-text">
      当前修订已变化、版本损坏或没有恢复权限。请保留理由，关闭后刷新版本与草稿再操作。
    </p>
    <p v-if="done" class="notice">草稿已恢复，尚未自动发布。</p>
    <p v-if="mutation.error.value" class="error-text" role="alert">{{ mutation.error.value }}</p>
    <div class="row">
      <el-button
        type="danger"
        :disabled="!allowed || !confirmed || !reason.trim()"
        :loading="mutation.pending.value"
        @click="restore"
      >
        确认覆盖并恢复草稿
      </el-button>
      <el-button :disabled="mutation.pending.value" @click="emit('cancel')">关闭恢复面板</el-button>
    </div>
  </section>
</template>

<script setup lang="ts">
/** 列表和详情显示管理状态；弃用或损坏版本不开放消费制品入口。 */
import { computed, ref, watch } from 'vue'
import type { Release, ReleaseDetail } from '../../../../../packages/api-client/src/release'
import { useWorkspace } from '../../stores/workspace'
import { useSession } from '../../stores/session'
import { useResource } from '../../composables/useResource'
import { useMutation, confirmAction } from '../../composables/useMutation'
import AsyncState from '@onto/shared-ui/AsyncState.vue'
import ArtifactList from './ArtifactList.vue'
import ReleaseEntities from './ReleaseEntities.vue'
import ReleaseDiffPanel from './ReleaseDiffPanel.vue'
import RestoreDraftPanel from './RestoreDraftPanel.vue'
import './p4.css'
const workspace = useWorkspace(),
  api = useSession().api,
  mutation = useMutation()
const releases = useResource<Release[]>([]),
  detail = useResource<ReleaseDetail | null>(null),
  document = useResource<unknown | null>(null)
const selected = ref(''),
  reason = ref(''),
  restoring = ref(false)
const owner = computed(() => workspace.space?.role === 'OWNER' && workspace.canWrite)
let generation = 0
async function load() {
  const id = workspace.space?.id
  if (!id || mutation.pending.value || restoring.value) return
  const result = await releases.load(() => api.get<Release[]>(`/releases?spaceId=${id}`))
  if (result) {
    if (!result.some((item) => item.id === selected.value)) selected.value = result[0]?.id ?? ''
    else await loadDetail()
  }
}
function loadDetail() {
  if (!selected.value) return
  const id = selected.value
  return detail.load(async () => {
    const value = await api.get<ReleaseDetail>(`/releases/${id}`)
    if (value.release.id !== id) throw new Error('版本详情与当前选择不符。')
    return value
  })
}
watch(selected, () => {
  generation++
  detail.reset()
  document.reset()
  reason.value = ''
  restoring.value = false
  void loadDetail()
})
watch(
  () => workspace.space?.id,
  () => {
    selected.value = ''
    releases.reset()
    detail.reset()
    void load()
  },
  { immediate: true },
)
async function action(kind: 'deprecate' | 'integrity-check') {
  const value = detail.data.value,
    ticket = generation,
    context = workspace.captureContext()
  if (
    !value ||
    !owner.value ||
    (kind === 'deprecate' && (value.release.status !== 'READY' || !reason.value.trim()))
  )
    return
  const releaseId = value.release.id
  if (
    kind === 'deprecate' &&
    (!(await confirmAction(`弃用 v${value.release.versionNo} 后消费者不能继续调用。确认弃用？`)) ||
      !workspace.isCurrentContext(context) ||
      ticket !== generation)
  )
    return
  await mutation.run(async ({ isCurrent }) => {
    const updated = await api.post<ReleaseDetail>(
      `/releases/${releaseId}/${kind}`,
      kind === 'deprecate' ? { reason: reason.value.trim() } : {},
    )
    if (!isCurrent() || ticket !== generation) return
    if (updated.release.id !== releaseId) throw new Error('版本操作返回了不同版本，请重新读取。')
    detail.data.value = updated
    document.reset()
    releases.data.value = releases.data.value.map((item) =>
      item.id === releaseId ? updated.release : item,
    )
  })
}
function readDocument(kind: 'manifest' | 'context-pack') {
  if (detail.data.value?.release.status !== 'READY') return
  return document.load(() => api.get<unknown>(`/releases/${selected.value}/${kind}`))
}
async function restored() {
  restoring.value = false
  await load()
}
</script>
<template>
  <div class="p4-page stack">
    <div class="page-heading">
      <div>
        <h1>正式版本</h1>
        <p class="muted">查看版本制品、冻结实体、版本差异与完整性。</p>
      </div>
      <el-button :disabled="mutation.pending.value || restoring" @click="load">刷新版本</el-button>
    </div>
    <p v-if="mutation.error.value" class="error-text" role="alert">{{ mutation.error.value }}</p>
    <AsyncState
      :loading="releases.loading.value"
      :error="releases.error.value"
      :empty="!releases.data.value.length"
      empty-text="尚无正式版本。请在候选发布中完成五门禁与制品准备。"
      @retry="load"
    >
      <div class="split">
        <section class="panel stack">
          <h2>版本目录</h2>
          <button
            v-for="item in releases.data.value"
            :key="item.id"
            class="selector record"
            :aria-pressed="selected === item.id"
            :disabled="mutation.pending.value || restoring"
            @click="selected = item.id"
          >
            <b>v{{ item.versionNo }} · {{ item.status }}</b>
            <p class="muted">{{ item.publishedAt }}</p>
            <span class="identity">{{ item.bundleSha256 }}</span>
          </button>
        </section>
        <AsyncState :loading="detail.loading.value" :error="detail.error.value" @retry="loadDetail">
          <section v-if="detail.data.value" class="panel stack">
            <h2>
              版本 v{{ detail.data.value.release.versionNo }} ·
              {{ detail.data.value.release.status }}
            </h2>
            <p v-if="detail.data.value.release.status !== 'READY'" class="readonly-note">
              当前版本不可消费；历史事实仍可管理浏览。
            </p>
            <dl class="details-grid">
              <dt>版本标识</dt>
              <dd>{{ detail.data.value.release.id }}</dd>
              <dt>发布人</dt>
              <dd>{{ detail.data.value.release.publishedBy }}</dd>

                {{ endpoint.opCode }} · {{ endpoint.status }}

                {{ endpoint.route }}

            <div class="row">

                :disabled="detail.data.value.release.status !== 'READY'"
                @click="readDocument('manifest')"
              >
                读取正式操作清单
              </el-button>
              <el-button
                :disabled="detail.data.value.release.status !== 'READY'"
                @click="readDocument('context-pack')"
              >
                读取正式上下文包
              </el-button>
            </div>
            <p v-if="document.error.value" class="error-text" role="alert">
              {{ document.error.value }}
            </p>
            <pre v-if="document.data.value">{{ JSON.stringify(document.data.value, null, 2) }}</pre>
            <ReleaseEntities :key="selected" :release-id="selected" />
            <template v-if="owner">
              <label>
                弃用理由
                <textarea
                  v-model="reason"
                  rows="2"
                  :disabled="mutation.pending.value || restoring"
                />
              </label>

                <el-button
                  :disabled="mutation.pending.value || restoring"
                  @click="action('integrity-check')"
                >
                  执行真实完整性检查
                </el-button>
                <el-button
                  type="danger"
                  :disabled="
                    mutation.pending.value ||
                    restoring ||
                    detail.data.value.release.status !== 'READY' ||
                    !reason.trim()
                  "

                  弃用此版本

                    restoring ||

      <RestoreDraftPanel

const kind = ref<ReleaseEntityKind | ''>(''),

import type { GateKind, Publication } from '../../../../../packages/api-client/src/release'

export const gateLabels: Record<GateKind, string> = {
  CONFLICT: '冲突',
  CONSISTENCY: '一致性',
  CONTRACT: '操作契约',
  TEST: '验收测试',
  REVIEW: '人工复核',
}
export const gateKinds = Object.keys(gateLabels) as GateKind[]
/** 豁免只能覆盖真实超子集的一致性结论，不能覆盖守卫、超时或执行失败。 */
export function canWaiveConsistency(value: Publication) {
  return value.checks.some(
    (check) =>
      check.kind === 'CONSISTENCY' &&
      check.candidateSha256 === value.candidateSha256 &&
      check.conclusion === 'NOT_CHECKED' &&
      check.reason === 'SUBSET_EXCEEDED',
  )
}
export function gatePassed(value: Publication, kind: GateKind) {
  const check = value.checks.find(
    (item) => item.kind === kind && item.candidateSha256 === value.candidateSha256,
  )
  return (
    check?.conclusion === 'PASSED' ||
    (kind === 'CONSISTENCY' &&
      canWaiveConsistency(value) &&
      value.waivers.some((item) => item.candidateSha256 === value.candidateSha256))
  )
}
export function allArtifactsReady(value: Publication) {
  return (['BUNDLE', 'MANIFEST', 'CONTEXT_PACK', 'KNOWLEDGE_INDEX'] as const).every((kind) =>
    value.artifacts.some(
      (item) =>
        item.kind === kind &&
        item.ready &&
        !!item.sha256 &&
        item.size !== null &&
        item.size >= 0 &&
        (kind !== 'BUNDLE' || item.sha256 === value.candidateSha256),
    ),
  )
}

  name: string
  type: 'string' | 'number' | 'boolean' | 'json'
  value: string
}
/** 业务参数逐字段维护，只有嵌套值才使用单字段 JSON；不混入身份或可信属性。 */
export function parameterObject(rows: ParameterRow[]): Record<string, unknown> {
  const result: Record<string, unknown> = {}
  for (const row of rows) {
    const name = row.name.trim()
    if (
      !name ||
      Object.hasOwn(result, name) ||
      ['__proto__', 'prototype', 'constructor'].includes(name)
    )
      throw new Error('参数名称不能为空、重复或使用保留名称。')
    if (row.type === 'number') {
      if (!row.value.trim() || !Number.isFinite(Number(row.value)))
        throw new Error(`${name} 必须为有效数值。`)
      result[name] = Number(row.value)
    } else if (row.type === 'boolean') {
      if (!['true', 'false'].includes(row.value)) throw new Error(`${name} 请选择 true 或 false。`)
      result[name] = row.value === 'true'
    } else if (row.type === 'json') {
      try {
        result[name] = JSON.parse(row.value) as unknown
      } catch {
        throw new Error(`${name} 必须为有效 JSON 值。`)
      }
    } else result[name] = row.value
  }
  return result
}
export function parameterRows(object: Record<string, unknown> | null): ParameterRow[] {
  return Object.entries(object ?? {}).map(([name, value]) => ({
    name,
    type:
      typeof value === 'string'
        ? 'string'
        : typeof value === 'number'
          ? 'number'
          : typeof value === 'boolean'
            ? 'boolean'
            : 'json',
    value: typeof value === 'string' ? value : JSON.stringify(value),
  }))
}

  if (!owner.value || item.status !== 'DEAD' || !reasons[item.id]?.trim()) return

    state.data.value = {
      ...state.data.value,
      events: state.data.value.events.map((row) => (row.id === item.id ? result : row)),
    }
  })
}

  spaceGrants: SpaceGrant[]

  rows = ref<ParameterRow[]>([]),

  const item = selected.value
  scope.value = item?.dataScope === null ? 'unlimited' : 'limited'

  expiresAt.value = item?.expiresAt
    ? new Date(
        new Date(item.expiresAt).getTime() - new Date(item.expiresAt).getTimezoneOffset() * 60000,

export * from './types'
export * from './supply'
export * from './data'
export * from './release'
export type { components, paths } from './generated'

/** 所有 HTTP、信封与网络失败共用此错误，页面据此提供刷新或重新登录。 */
export class ApiError extends Error {
  constructor(
    public readonly code: string,
    message: string,
    public readonly status: number,
  ) {
    super(message)
    this.name = 'ApiError'
  }
}
export interface Envelope<T> {
  code: string
  message?: string
  data: T
  meta?: unknown
}
export class ApiClient {
  constructor(
    private readonly token: () => string | null,
    private readonly expired: () => void,
    private readonly fetcher: typeof fetch = fetch,
  ) {}
  /** 令牌仅在请求时读取内存；不接触 localStorage/sessionStorage。 */
  async request<T>(path: string, method = 'GET', body?: unknown): Promise<T> {
    return (await this.requestEnvelope<T>(path, method, body)).data
  }
  /** 内部候选调用需保留 preview 与摘要元数据，业务结果仍独立置于 data。 */
  async requestEnvelope<T>(path: string, method = 'GET', body?: unknown): Promise<Envelope<T>> {
    const headers = new Headers({ Accept: 'application/json' })
    const token = this.token()
    if (token) headers.set('Authorization', `Bearer ${token}`)
    // multipart boundary 由浏览器生成；否则受管文件会被误序列化为空 JSON。
    const multipart = body instanceof FormData
    if (body !== undefined && !multipart) headers.set('Content-Type', 'application/json')
    let response: Response
    try {
      // 原生 fetch 按全局函数调用，避免实例代理改变浏览器要求的调用上下文。
      const fetcher = this.fetcher
      response = await fetcher(`/onto${path}`, {
        method,
        headers,
        body: body === undefined ? undefined : multipart ? body : JSON.stringify(body),
      })
    } catch {
      throw new ApiError('NETWORK_ERROR', '无法连接服务，请检查网络后重试。', 0)
    }
    // 旧会话的迟到响应只能使旧请求失败，不能注销用户重新登录后的新会话。
    if (response.status === 401 && token === this.token()) this.expired()
    let envelope: Envelope<T>
    try {
      const parsed: unknown = await response.json()
      if (
        typeof parsed !== 'object' ||
        parsed === null ||
        !('code' in parsed) ||
        typeof parsed.code !== 'string' ||
        ('message' in parsed && typeof parsed.message !== 'string') ||
        (parsed.code !== 'OK' && !('message' in parsed)) ||
        !('data' in parsed)
      )
        throw new Error('Invalid envelope')
      envelope = parsed as Envelope<T>
    } catch {
      throw new ApiError('INVALID_RESPONSE', '服务响应无法读取，请稍后重试。', response.status)
    }
    if (!response.ok || envelope.code !== 'OK')
      throw new ApiError(
        envelope.code || 'HTTP_ERROR',
        envelope.message ||
          (response.status === 403 ? '当前账号没有此操作权限。' : '操作失败，请重试。'),
        response.status,
      )
    return envelope
  }
  get<T>(path: string) {
    return this.request<T>(path)
  }
  post<T>(path: string, body: unknown) {
    return this.request<T>(path, 'POST', body)
  }
  put<T>(path: string, body: unknown) {
    return this.request<T>(path, 'PUT', body)
  }
  delete<T>(path: string) {
    return this.request<T>(path, 'DELETE')
  }
}
