import { watch } from 'vue'
import type { MetadataSnapshot, MetadataSummary } from '@onto/api-client'
import { useSession } from '../../stores/session'
import { useWorkspace } from '../../stores/workspace'
import { useResource } from '../../composables/useResource'

/** 切源立即撤销旧快照请求；相同快照 ID 也必须同时归属于所选源和空间。 */
export function useMetadataSelection(sourceId: () => string, snapshotId: () => string) {
  const api = useSession().api,
    workspace = useWorkspace()
  const summaries = useResource<MetadataSummary[]>([]),
    snapshot = useResource<MetadataSnapshot | null>(null)
  async function loadSummaries() {
    const source = sourceId(),
      spaceId = workspace.space?.id
    if (source && spaceId)
      await summaries.load(() =>
        api.get<MetadataSummary[]>(`/datasources/${source}/snapshots?spaceId=${spaceId}`),
      )
  }
  async function loadSnapshot() {
    const source = sourceId(),
      id = snapshotId(),
      spaceId = workspace.space?.id
    if (source && id && spaceId)
      await snapshot.load(() =>
        api.get<MetadataSnapshot>(`/datasources/${source}/snapshots/${id}?spaceId=${spaceId}`),
      )
  }
  watch(
    sourceId,
    () => {
      summaries.reset()
      snapshot.reset()
      void loadSummaries()
    },
    { immediate: true },
  )
  watch(
    [sourceId, snapshotId],
    () => {
      snapshot.reset()
      void loadSnapshot()
    },
    { immediate: true },
  )
  return { summaries, snapshot, loadSummaries, loadSnapshot }
}

/** 默认既不勾选也不确认；显式接受/拒绝后才提交原子批次，409 保留已编辑载荷。 */
import { computed, reactive, ref } from 'vue'
import type { OntClass, Suggestion, SuggestionDecision } from '@onto/api-client'

  props.items.map((item) => ({
    ...item,
    payload: { ...item.payload },
    checked: false,
    decision: '' as '' | 'ACCEPT' | 'REJECT',
  })),
)

  disabled = computed(() => !props.canWrite || props.pending || props.stale)

function bulk(decision: 'ACCEPT' | 'REJECT') {
  if (disabled.value) return
  for (const row of rows) if (row.checked && row.status === 'PENDING') row.decision = decision
}
function submit() {
  error.value = ''
  if (disabled.value) return
  const chosen = rows.filter((item) => item.status === 'PENDING' && item.decision)
  if (!chosen.length) {
    error.value = '请先选择候选的接受或拒绝决定。'
    return
  }
  if (
    chosen.some(
      (item) =>
        item.decision === 'ACCEPT' &&
        (typeof item.payload.name !== 'string' || !item.payload.name.trim()),
    )
  ) {
    error.value = '接受候选必须填写名称。'
    return
  }
  emit(
    'decide',
    chosen.map((item) =>
      item.decision === 'ACCEPT'
        ? { id: item.id, decision: 'ACCEPT', payload: { ...item.payload } }
        : { id: item.id, decision: 'REJECT' },
    ),
  )
}

/** 名称、类型与关联以基本字段维护；候选关联 ID 与实体 ID 互斥提交。 */
import { computed } from 'vue'
import type { JsonObject, OntClass, Suggestion } from '@onto/api-client'
const props = defineProps<{
  kind: Suggestion['kind']
  classes: OntClass[]
  suggestions: Suggestion[]
  batchId: string
  suggestionId: string
  disabled: boolean
}>()
const payload = defineModel<JsonObject>({ required: true })
const classCandidates = computed(() =>
  props.suggestions.filter(
    (item) =>
      item.kind === 'CLASS' &&
      item.id !== props.suggestionId &&
      item.batchId === props.batchId &&
      item.status !== 'REJECTED',
  ),
)
function text(key: string) {
  const value = payload.value[key]
  return typeof value === 'string' ? value : ''
}
function set(key: string, value: unknown) {
  payload.value = { ...payload.value, [key]: value }
}
function reference(entity: string, candidate: string) {
  return text(entity)
    ? `entity:${text(entity)}`
    : text(candidate)
      ? `suggestion:${text(candidate)}`
      : ''
}
function setReference(entity: string, candidate: string, value: string) {
  const next = { ...payload.value }
  delete next[entity]
  delete next[candidate]
  if (value.startsWith('entity:')) next[entity] = value.slice(7)
  else if (value.startsWith('suggestion:')) next[candidate] = value.slice(11)
  else if (entity === 'parentId') next.parentId = null
  payload.value = next
}
const associations = computed(() =>
  props.kind === 'CLASS'
    ? [{ label: '父类（可选）', entity: 'parentId', candidate: 'parentSuggestionId' }]
    : props.kind === 'PROPERTY'
      ? [{ label: '所属类', entity: 'domainClassId', candidate: 'domainSuggestionId' }]
      : [
          { label: '起点类', entity: 'fromClassId', candidate: 'fromSuggestionId' },
          { label: '终点类', entity: 'toClassId', candidate: 'toSuggestionId' },
        ],
)

<script setup lang="ts">
/** OWNER 专用配置表单：读取 DTO 不含凭据；null 明确保留已有连接。 */
import { computed, reactive, ref, watch } from 'vue'
import type { Datasource, ManagedFile, SourceKind, SourceWrite } from '@onto/api-client'
import { connectionFields, dialects, lines, objectJson, sourceLabels } from './sourceFields'
const props = defineProps<{
  value: Datasource | null
  files?: ManagedFile[]
  canWrite: boolean
  pending: boolean
  stale: boolean
  error?: string
}>()
const emit = defineEmits<{ save: [value: SourceWrite]; cancel: [] }>()
const fields = reactive({
  code: '',
  name: '',
  kind: 'RELATIONAL' as SourceKind,
  scope: '',
  dialect: 'postgresql',
  endpoints: '',
  headers: '',
})
const connection = reactive<Record<string, string>>({}),
  replace = ref(false),
  localError = ref('')
watch(
  () => props.value,
  (value) => {
    Object.assign(fields, {
      code: value?.code ?? '',
      name: value?.name ?? '',
      kind: value?.kind ?? 'RELATIONAL',
      scope: value?.accessScope.join('\n') ?? '',
      dialect: value?.dialect || 'postgresql',
      endpoints: '',
      headers: '',
    })
    for (const key of Object.keys(connection)) delete connection[key]
    replace.value = !value
    localError.value = ''
  },
  { immediate: true },
)
watch(
  () => fields.kind,
  (kind) => {
    fields.dialect = dialects[kind]?.[0] ?? ''
  },
)
watch(
  () => connection.fileId,
  (id) => {
    const file = props.files?.find((item) => item.id === id)
    if (file) {
      fields.scope = file.resource
      fields.dialect = /\.xlsx$/i.test(file.name)
        ? 'excel'
        : /\.json$/i.test(file.name)
          ? 'json'
          : 'csv'
    }
  },
)
const disabled = computed(() => !props.canWrite || props.pending || props.stale)
function save() {
  localError.value = ''
  if (disabled.value) return
  try {
    if (!fields.code.trim() || !fields.name.trim() || !lines(fields.scope).length)
      throw new Error('编码、名称与访问范围必填。')
    let config: Record<string, unknown> | null = null
    if (replace.value) {
      config = { dialect: fields.dialect }
      for (const field of connectionFields[fields.kind]) {
        // 凭据是精确字节语义，密码首尾空格不能被表单擅自规范化。
        const raw = connection[field.key] ?? ''
        const value = field.secret ? raw : raw.trim()
        if (field.required && !value.trim()) throw new Error(`${field.label}必填。`)
        if (value)
          config[field.key] =
            field.key === 'contactPoints'
              ? lines(value)
              : field.key === 'port' || (fields.kind === 'KEY_VALUE' && field.key === 'database')
                ? Number(value)
                : value
      }
      if (fields.kind === 'FILE') {
        const file = props.files?.find((item) => item.id === connection.fileId)
        const scope = lines(fields.scope)
        if (!file || scope.length !== 1 || scope[0] !== file.resource)
          throw new Error('受管文件访问范围必须精确匹配所选文件的服务端资源名。')
      }
      if (fields.kind === 'HTTP') {
        config.endpoints = lines(fields.endpoints)
        const headers = objectJson(fields.headers, '请求头')
        if (headers) config.headers = headers
      }
    }
    emit('save', {
      code: fields.code.trim(),
      name: fields.name.trim(),
      kind: fields.kind,
      connection: config,
      accessScope: lines(fields.scope),
    })
  } catch (cause) {
    localError.value = cause instanceof Error ? cause.message : '请检查源配置。'
  }
}

export const dialects: Partial<Record<SourceKind, string[]>> = {
  RELATIONAL: ['postgresql'],
  WAREHOUSE: ['clickhouse'],
  LAKE: ['s3-parquet'],
  FILE: ['csv', 'json', 'excel'],
  HTTP: ['json'],
  DOCUMENT: ['mongodb'],
  KEY_VALUE: ['redis'],
  WIDE_COLUMN: ['cassandra'],
  GRAPH: ['neo4j'],
}
export function lines(text: string) {
  return text
    .split(/[\n,]/)
    .map((item) => item.trim())
    .filter(Boolean)
}
export function objectJson(text: string, field: string): Record<string, unknown> | null {
  if (!text.trim()) return null
  const value: unknown = JSON.parse(text)
  if (!value || Array.isArray(value) || typeof value !== 'object')
    throw new Error(`${field}必须是 JSON 对象`)
  return value as Record<string, unknown>
}

<script setup lang="ts">
/** 文件输入只形成 multipart；8 MiB 和扩展名仅提前反馈，内容判定仍由服务端执行。 */
import { ref } from 'vue'
import type { ManagedFile } from '@onto/api-client'
defineProps<{ canWrite: boolean; pending: boolean; stale: boolean; files: ManagedFile[] }>()
const emit = defineEmits<{ upload: [file: File] }>(),
  error = ref('')
function choose(event: Event) {
  error.value = ''
  const input = event.target as HTMLInputElement,
    file = input.files?.[0]
  if (!file) return
  if (file.size > 8 * 1024 * 1024) error.value = '文件不能超过 8 MiB。'
  else if (!/\.(csv|json|xlsx|pdf|docx|txt)$/i.test(file.name))
    error.value = '仅允许 CSV / JSON / XLSX / PDF / DOCX / TXT。'
  else emit('upload', file)
  input.value = ''
}
</script>
<template>
  <section class="panel stack">
    <h2>受管文件上传</h2>
    <p class="muted">
      OWNER 可上传最大 8 MiB 文件。结构文件用于映射与抽取，文档用于建模抽取；服务端校验真实内容。
    </p>
    <label>
      选择文件
      <input
        type="file"
        accept=".csv,.json,.xlsx,.pdf,.docx,.txt"
        :disabled="!canWrite || pending || stale"
        @change="choose"
      />
    </label>
    <p v-if="error" class="error-text" role="alert">{{ error }}</p>
    <dl v-for="file in files" :key="file.id" class="details-grid">
      <dt>{{ file.name }}</dt>
      <dd>
        {{ file.size }} bytes ·
        <code>{{ file.id }}</code>
        <br />
        访问范围：

    emit('save', null)

export * from './types'

/** 所有 HTTP、信封与网络失败共用此错误，页面据此提供刷新或重新登录。 */
export class ApiError extends Error {
  constructor(
    public readonly code: string,
    message: string,
    public readonly status: number,
  ) {
    super(message)
    this.name = 'ApiError'
  }
}
export interface Envelope<T> {
  code: string
  message?: string
  data: T
  meta?: unknown
}
export class ApiClient {
  constructor(
    private readonly token: () => string | null,
    private readonly expired: () => void,
    private readonly fetcher: typeof fetch = fetch,
  ) {}
  /** 令牌仅在请求时读取内存；不接触 localStorage/sessionStorage。 */
  async request<T>(path: string, method = 'GET', body?: unknown): Promise<T> {
    return (await this.requestEnvelope<T>(path, method, body)).data
  }
  /** 内部候选调用需保留 preview 与摘要元数据，业务结果仍独立置于 data。 */
  async requestEnvelope<T>(path: string, method = 'GET', body?: unknown): Promise<Envelope<T>> {
    const headers = new Headers({ Accept: 'application/json' })
    const token = this.token()
    if (token) headers.set('Authorization', `Bearer ${token}`)
    // multipart boundary 由浏览器生成；否则受管文件会被误序列化为空 JSON。
    const multipart = body instanceof FormData
    if (body !== undefined && !multipart) headers.set('Content-Type', 'application/json')
    let response: Response
    try {
      // 原生 fetch 按全局函数调用，避免实例代理改变浏览器要求的调用上下文。
      const fetcher = this.fetcher
      response = await fetcher(`/onto${path}`, {
        method,
        headers,
        body: body === undefined ? undefined : multipart ? body : JSON.stringify(body),
      })
    } catch {
      throw new ApiError('NETWORK_ERROR', '无法连接服务，请检查网络后重试。', 0)
    }
    // 旧会话的迟到响应只能使旧请求失败，不能注销用户重新登录后的新会话。
    if (response.status === 401 && token === this.token()) this.expired()
    let envelope: Envelope<T>
    try {
      const parsed: unknown = await response.json()
      if (
        typeof parsed !== 'object' ||
        parsed === null ||
        !('code' in parsed) ||
        typeof parsed.code !== 'string' ||
        ('message' in parsed && typeof parsed.message !== 'string') ||
        (parsed.code !== 'OK' && !('message' in parsed)) ||
        !('data' in parsed)
      )
        throw new Error('Invalid envelope')
      envelope = parsed as Envelope<T>
    } catch {
      throw new ApiError('INVALID_RESPONSE', '服务响应无法读取，请稍后重试。', response.status)
    }
    if (!response.ok || envelope.code !== 'OK')
      throw new ApiError(
        envelope.code || 'HTTP_ERROR',
        envelope.message ||
          (response.status === 403 ? '当前账号没有此操作权限。' : '操作失败，请重试。'),
        response.status,
      )
    return envelope
  }
  get<T>(path: string) {
    return this.request<T>(path)
  }
  post<T>(path: string, body: unknown) {
    return this.request<T>(path, 'POST', body)
  }
  put<T>(path: string, body: unknown) {
    return this.request<T>(path, 'PUT', body)
  }
  delete<T>(path: string) {
    return this.request<T>(path, 'DELETE')
  }
}
