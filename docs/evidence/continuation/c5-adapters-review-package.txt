        new OutputFile() {

              public long getPos() {

              }

              public void write(int b) {
                bytes.write(b);
              }

              public void write(byte[] b, int o, int n) {
                bytes.write(b, o, n);
              }
            };
          }

          public PositionOutputStream createOrOverwrite(long hint) {
            return create(hint);
          }

          public boolean supportsBlockSize() {
            return false;
          }

          public long defaultBlockSize() {
            return 0;
          }
        };
    var schema =
        MessageTypeParser.parseMessageType(
            "message rows { required int32 id; optional binary region (STRING); optional int64"
                + " amount; optional binary hidden (STRING); }");
    try (var writer =
        ExampleParquetWriter.builder(output)
            .withConf(new Configuration(false))
            .withType(schema)
            .withCompressionCodec(CompressionCodecName.UNCOMPRESSED)
            .build()) {
      writer.write(
          new SimpleGroup(schema).append("id", 1).append("region", "B").append("amount", 10L));
      writer.write(new SimpleGroup(schema).append("id", 2).append("amount", 20L));
      writer.write(
          new SimpleGroup(schema)
              .append("id", 3)
              .append("region", "A")
              .append("amount", 30L)
              .append("hidden", "private"));
    }
    byte[] body = bytes.toByteArray();
    assertEquals("PAR1", new String(body, 0, 4, java.nio.charset.StandardCharsets.US_ASCII));
    client.putObject(
        PutObjectArgs.builder().bucket(config.path("bucket").asText()).object(object).stream(
                new ByteArrayInputStream(body), (long) body.length, -1L)
            .build());

      assertEquals(1, adapter.metadata(source).resources().size());
      var q =
          new QuerySpec(
              object,
              List.of("id", "amount"),

              List.of(),
              1,
              10000,
              Map.of(),
              LabelScope.parse(mapper.readTree("{\"attr\":\"region\",\"in\":[\"A\"]}")),
              Map.of("region", "region"));

      assertEquals(30L, result.rows().getFirst().get("amount"));

      assertThrows(
          DataProblem.class,
          () -> adapter.query(new SourceSpec("lake", "LAKE", config, Set.of("other/")), q));

  }

        predicate = and(predicate, in(schema, b.getValue(), values));
      }
    }
    for (var f : q.filters()) {
      if (f.operator() != QuerySpec.Operator.EQ && f.operator() != QuerySpec.Operator.IN)
        throw QueryPlanner.problem("PARQUET_OPERATOR_UNSUPPORTED");
      if (f.values().isEmpty()) return new QueryResult(meta.fields(), List.of(), List.of());
      predicate = and(predicate, in(schema, f.column(), f.values()));

      long ordinal = 0;

          if (row.getFieldRepetitionCount(index) > 0)

                  case INT64 -> row.getLong(index, 0);
                  case FLOAT -> row.getFloat(index, 0);
                  case DOUBLE -> row.getDouble(index, 0);

                  case BINARY -> row.getBinary(index, 0).toStringUsingUTF8();

  private FilterPredicate and(FilterPredicate a, FilterPredicate b) {
    return a == null ? b : FilterApi.and(a, b);
  }

  private FilterPredicate in(
      MessageType schema, String field, List<com.fasterxml.jackson.databind.JsonNode> values) {
    FilterPredicate result = null;
    for (var value : values) {
      FilterPredicate next =
          switch (schema.getType(field).asPrimitiveType().getPrimitiveTypeName()) {
            case INT32 -> FilterApi.eq(FilterApi.intColumn(field), value.intValue());
            case INT64 -> FilterApi.eq(FilterApi.longColumn(field), value.longValue());
            case FLOAT -> FilterApi.eq(FilterApi.floatColumn(field), value.floatValue());
            case DOUBLE -> FilterApi.eq(FilterApi.doubleColumn(field), value.doubleValue());
            case BOOLEAN -> FilterApi.eq(FilterApi.booleanColumn(field), value.booleanValue());
            case BINARY ->
                FilterApi.eq(FilterApi.binaryColumn(field), Binary.fromString(value.asText()));
            default -> throw QueryPlanner.problem("PARQUET_TYPE_UNSUPPORTED");
          };
      result = result == null ? next : FilterApi.or(result, next);
    }
    return result;
  }

  static final class BytesFile implements InputFile {
    private final byte[] bytes;

    BytesFile(byte[] bytes) {
      this.bytes = bytes;
    }

    @Override
    public long getLength() {
      return bytes.length;
    }

    @Override
    public SeekableInputStream newStream() {
      return new SeekableInputStream() {
        int pos;

        @Override
        public long getPos() {
          return pos;
        }

        @Override
        public void seek(long p) throws IOException {
          if (p < 0 || p > bytes.length) throw new EOFException();
          pos = (int) p;
        }

        @Override
        public int read() {
          return pos == bytes.length ? -1 : bytes[pos++] & 255;
        }

        @Override
        public int read(byte[] b, int off, int len) {
          if (pos == bytes.length) return -1;
          int n = Math.min(len, bytes.length - pos);
          System.arraycopy(bytes, pos, b, off, n);
          pos += n;
          return n;
        }

        @Override
        public void readFully(byte[] b) throws IOException {
          readFully(b, 0, b.length);
        }

        @Override
        public void readFully(byte[] b, int off, int len) throws IOException {
          if (bytes.length - pos < len) throw new EOFException();
          System.arraycopy(bytes, pos, b, off, len);
          pos += len;
        }

        @Override
        public int read(ByteBuffer b) {
          if (pos == bytes.length) return -1;
          int n = Math.min(b.remaining(), bytes.length - pos);
          b.put(bytes, pos, n);
          pos += n;
          return n;
        }

        @Override
        public void readFully(ByteBuffer b) throws IOException {
          if (b.remaining() > bytes.length - pos) throw new EOFException();
          int n = b.remaining();
          b.put(bytes, pos, n);
          pos += n;
        }
      };
    }

              out.add(
                  new SourceMetadata.Resource(
                      r, inferred.fields(), List.of("_key"), List.of(), List.of()));
              found = true;
              break;
            }
          }
          if (found || cursor.equals("0")) break;
          if (step == 999) throw QueryPlanner.problem("SCAN_LIMIT_EXCEEDED");
        }
        if (!found) throw QueryPlanner.problem("SCOPE_UNAVAILABLE");
      }
      return new SourceMetadata(out);
    }
  }

  @Override
  QueryResult execute(SourceSpec s, QuerySpec q) throws Exception {
    flat(q);
    prefix(s, q.resource());
    long started = System.nanoTime();
    List<Map<String, Object>> out = new ArrayList<>();
    List<QueryResult.SourceRecord> evidence = new ArrayList<>();
    String exact = null;
    for (var f : q.filters())
      if (f.column().equals("_key") && f.operator() == QuerySpec.Operator.EQ) {
        exact = f.values().getFirst().asText();
        if (!exact.startsWith(q.resource())) throw QueryPlanner.problem("SCOPE_EXCEEDED");
      }
    try (Jedis j = open(s)) {
      String cursor = "0";
      int scanned = 0;
      for (int step = 0; step < 1000; step++) {
        deadline(started, q);
        ScanResult<String> scan =
            exact == null
                ? j.scan(cursor, new ScanParams().match(q.resource() + "*").count(100))
                : new ScanResult<>("0", List.of(exact));
        cursor = scan.getCursor();
        for (String key : scan.getResult()) {
          if (++scanned > MAX_SCAN) throw QueryPlanner.problem("SCAN_LIMIT_EXCEEDED");
          if (!key.startsWith(q.resource())) throw QueryPlanner.problem("SCOPE_EXCEEDED");
          if (!j.exists(key)) continue;
          var row = row(j, key);
          if (row != null) {
            fields(q, row.keySet());
            if (matches(q, row)) {
              out.add(project(q, row));
              evidence.add(new QueryResult.SourceRecord(s.id(), q.resource(), Map.of("key", key)));
              if (out.size() >= q.limit())
                return new QueryResult(inferred(q.resource(), out).fields(), out, evidence);
            }
          }
        }
        if (cursor.equals("0"))
          return new QueryResult(inferred(q.resource(), out).fields(), out, evidence);
      }
      throw QueryPlanner.problem("SCAN_LIMIT_EXCEEDED");
    }
  }
}

/** 标签严格标识符；只读会话执行自动生成的节点查询，禁止用户 Cypher。 */
public final class Neo4jSourceAdapter extends AdapterSupport {
  record GraphPlan(String cypher, Map<String, Object> parameters) {}

  GraphPlan plan(SourceSpec s, QuerySpec q) {
    flat(q, Set.of("relationship", "targetLabel"));
    if (!q.options().keySet().equals(Set.of("relationship", "targetLabel")))
      throw QueryPlanner.problem("INVALID_GRAPH_RELATIONSHIP");
    String source = label(s, q.resource()),
        target = label(s, q.options().get("targetLabel").asText()),
        type = q.options().get("relationship").asText();
    QueryPlanner.identifier(type);
    if (type.contains(".")) throw QueryPlanner.problem("INVALID_IDENTIFIER");
    List<String> where = new ArrayList<>();
    Map<String, Object> parameters = new LinkedHashMap<>();
    for (var b : q.guardBindings().entrySet())
      for (String alias : List.of("n", "m")) {
        String column = property(b.getValue()).replace("n.", alias + ".");
        where.add(column + " IS NOT NULL AND trim(toString(" + column + ")) <> ''");
        var values = q.scope().parameters().get(b.getKey());
        if (values != null) {
          String p = "p" + parameters.size();
          parameters.put(p, values.stream().map(this::nativeValue).toList());
          where.add(column + " IN $" + p);
        }
      }
    if (!q.filters().isEmpty()) throw QueryPlanner.problem("GRAPH_EDGE_FILTER_UNSUPPORTED");
    List<String> projection = new ArrayList<>();
    for (String p : q.projection())
      projection.add(
          switch (p) {
            case "_targetId" -> "elementId(m) AS `_targetId`";
            case "_relationshipId" -> "elementId(r) AS `_relationshipId`";
            default -> property(p) + " AS `" + p + "`";
          });
    parameters.put("limit", q.limit());
    return new GraphPlan(
        "MATCH (n:"
            + source
            + ")-[r:`"
            + type
            + "`]->(m:"
            + target
            + ") WHERE "
            + String.join(" AND ", where)
            + " RETURN "
            + String.join(",", projection)
            + ",elementId(n) AS __key,elementId(m) AS __target,elementId(r) AS __relation LIMIT"
            + " $limit",
        parameters);
  }

  @Override
  public Set<String> capabilities() {
    return Set.of(
        "NODE_LABEL_QUERY",
        "SINGLE_HOP_RELATIONSHIP",
        "PARAMETERIZED_FILTER",
        "READ_TRANSACTION",
        "SAMPLED_PROPERTIES",
        "ELEMENT_ID_EVIDENCE",
        "NO_JOIN");

  Object nativeValue(com.fasterxml.jackson.databind.JsonNode n) {
    if (n.isIntegralNumber()) {
      if (!n.canConvertToLong()) throw QueryPlanner.problem("GRAPH_NUMBER_RANGE");
      return n.longValue();
    }
    if (n.isNumber()) {
      double v = n.doubleValue();
      if (!Double.isFinite(v) || java.math.BigDecimal.valueOf(v).compareTo(n.decimalValue()) != 0)
        throw QueryPlanner.problem("GRAPH_NUMBER_PRECISION");
      return v;
    }
    return QueryPlanner.scalar(n);
  }

  QueryResult executeRelationship(SourceSpec s, QuerySpec q) {
    var plan = plan(s, q);
    try (Driver d = open(s);
        Session session = session(d, s)) {
      return session.executeRead(
          tx -> {
            List<Map<String, Object>> rows = new ArrayList<>();

            var cursor = tx.run(plan.cypher(), plan.parameters());
            while (cursor.hasNext() && rows.size() < q.limit()) {
              var record = cursor.next();
              Map<String, Object> row = new LinkedHashMap<>();

              rows.add(row);

                          "targetElementId",
                          record.get("__target").asString(),
                          "relationshipElementId",
                          record.get("__relation").asString())));

            return new QueryResult(inferred(q.resource(), rows).fields(), rows, evidence);
          },
          TransactionConfig.builder().withTimeout(Duration.ofMillis(q.timeoutMs())).build());

                  if (!result.hasNext()) throw QueryPlanner.problem("SCOPE_UNAVAILABLE");

    QueryPlanner.identifier(field);
    if (field.contains(".")) throw QueryPlanner.problem("INVALID_IDENTIFIER");
    return "n.`" + field + "`";
  }

  @Override
  QueryResult execute(SourceSpec s, QuerySpec q) {
    if (!q.options().isEmpty()) return executeRelationship(s, q);
    flat(q);
    String label = label(s, q.resource());
    List<String> predicates = new ArrayList<>();
    Map<String, Object> params = new LinkedHashMap<>();
    for (var b : q.guardBindings().entrySet()) {
      String c = property(b.getValue());
      predicates.add(c + " IS NOT NULL AND trim(toString(" + c + ")) <> ''");
      var allowed = q.scope().parameters().get(b.getKey());
      if (allowed != null) {
        String p = "p" + params.size();
        params.put(p, allowed.stream().map(this::nativeValue).toList());
        predicates.add(c + " IN $" + p);
      }
    }
    for (var f : q.filters()) {
      String p = "p" + params.size();
      Object value =
          f.operator() == QuerySpec.Operator.IN
              ? f.values().stream().map(this::nativeValue).toList()
              : nativeValue(f.values().getFirst());
      if (value instanceof java.math.BigDecimal decimal) value = decimal.doubleValue();
      params.put(p, value);
      predicates.add(
          property(f.column())
              + " "
              + switch (f.operator()) {
                case EQ -> "=";
                case IN -> "IN";
                case GT -> ">";
                case GE -> ">=";
                case LT -> "<";
                case LE -> "<=";
              }
              + " $"
              + p);
    }
    params.put("limit", q.limit());
    String projection =
        q.projection().stream()
            .map(p -> property(p) + " AS `" + p + "`")
            .collect(java.util.stream.Collectors.joining(","));
    String cypher =
        "MATCH (n:"
            + label
            + ") WHERE "
            + String.join(" AND ", predicates)
            + " RETURN "
            + projection
            + ", elementId(n) AS __key LIMIT $limit";
    try (Driver d = open(s);
        Session session = session(d, s)) {
      return session.executeRead(
          tx -> {
            List<Map<String, Object>> rows = new ArrayList<>();
            List<QueryResult.SourceRecord> evidence = new ArrayList<>();
            var cursor = tx.run(cypher, params);
            while (cursor.hasNext() && rows.size() < q.limit()) {
              var record = cursor.next();
              Map<String, Object> row = new LinkedHashMap<>();
              q.projection().forEach(p -> row.put(p, record.get(p).asObject()));
              rows.add(row);
              evidence.add(
                  new QueryResult.SourceRecord(
                      s.id(), q.resource(), Map.of("elementId", record.get("__key").asString())));
            }
            return new QueryResult(inferred(q.resource(), rows).fields(), rows, evidence);
          },
          TransactionConfig.builder().withTimeout(Duration.ofMillis(q.timeoutMs())).build());
    }
  }

  }

      List<SourceMetadata.Resource> out = new ArrayList<>();
      for (String r : s.accessScope()) {
        var coll = c.getDatabase(text(s, "database")).getCollection(collection(s, r));
        Document doc = coll.find().maxTime(3, TimeUnit.SECONDS).first();
        List<SourceMetadata.Index> indexes = new ArrayList<>();
        for (Document idx : coll.listIndexes().maxTime(3, TimeUnit.SECONDS)) {
          Document keys = idx.get("key", Document.class);
          indexes.add(

                  Boolean.TRUE.equals(idx.getBoolean("unique"))));
        }

        out.add(
            new SourceMetadata.Resource(r, sampled.fields(), List.of("_id"), List.of(), indexes));
      }

    Object v = QueryPlanner.scalar(n);
    return v instanceof java.math.BigDecimal d ? new Decimal128(d) : v;

      String field = b.getValue();
      predicates.add(Filters.exists(field, true));
      predicates.add(
          Filters.expr(

                      new Document("$type", "$" + field),
                      List.of("string", "int", "long", "double", "decimal", "bool")))));

      predicates.add(

                                  "$convert",

                                      .append("onError", "")
                                      .append("onNull", "")))),

      List<Object> v = f.values().stream().map(MongoSourceAdapter::value).toList();
      predicates.add(
          switch (f.operator()) {
            case EQ -> Filters.eq(f.column(), v.getFirst());
            case IN -> Filters.in(f.column(), v);
            case GT -> Filters.gt(f.column(), v.getFirst());
            case GE -> Filters.gte(f.column(), v.getFirst());
            case LT -> Filters.lt(f.column(), v.getFirst());
            case LE -> Filters.lte(f.column(), v.getFirst());
          });
    }
    try (MongoClient c = open(s)) {
      var coll = c.getDatabase(text(s, "database")).getCollection(name);
      List<Map<String, Object>> out = new ArrayList<>();
      List<QueryResult.SourceRecord> evidence = new ArrayList<>();

              .limit(q.limit())
              .maxTime(q.timeoutMs(), TimeUnit.MILLISECONDS)

          Document d = cursor.next();
          out.add(project(q, d));
          evidence.add(
              new QueryResult.SourceRecord(s.id(), q.resource(), Map.of("_id", d.get("_id"))));
        }
      }
      return new QueryResult(inferred(q.resource(), out).fields(), out, evidence);
    }

import cn.ontorepo.data.domain.*;

import java.util.*;

/** PostgreSQL / ClickHouse 明确分派，所有实例查询只使用结构化计划。 */
public final class JdbcSourceAdapter extends AdapterSupport {
  @Override
  public Set<String> capabilities() {

        "METADATA",

        "INNER_EQ_JOIN",
        "SCOPE_BEFORE_LIMIT",
        "PRIMARY_KEY_EVIDENCE");

  @Override
  void connectivity(SourceSpec s) throws Exception {
    String url = text(s, "url");
    uriSocket(url.substring(5), s.kind().equals("WAREHOUSE") ? 8123 : 5432);
  }

  Connection open(SourceSpec s) throws Exception {
    String dialect = text(s, "dialect"), url = text(s, "url");
    if (!(s.kind().equals("RELATIONAL")
            && dialect.equals("postgresql")
            && url.startsWith("jdbc:postgresql://")
        || s.kind().equals("WAREHOUSE")
            && dialect.equals("clickhouse")
            && url.startsWith("jdbc:clickhouse://")))
      throw QueryPlanner.problem("UNSUPPORTED_DIALECT");
    Properties p = new Properties();
    p.setProperty("user", text(s, "user"));
    p.setProperty("password", s.connection().path("password").asText(""));
    p.setProperty("connectTimeout", "3");
    p.setProperty("socketTimeout", "10");
    return DriverManager.getConnection(url, p);
  }

  @Override
  void authenticate(SourceSpec s) throws Exception {
    try (Connection c = open(s)) {
      if (!c.isValid(3)) throw QueryPlanner.problem("AUTHENTICATION_FAILED");
    }
  }

  @Override
  SourceMetadata readMetadata(SourceSpec s) throws Exception {
    try (Connection c = open(s)) {
      if (s.kind().equals("WAREHOUSE")) return clickHouseMetadata(c, s);
      DatabaseMetaData m = c.getMetaData();
      List<SourceMetadata.Resource> out = new ArrayList<>();
      for (String r : s.accessScope()) {
        QueryPlanner.identifier(r);
        String[] names = r.split("\\.");
        if (names.length != 2) throw QueryPlanner.problem("INVALID_RESOURCE");
        String schema = names[0], table = names[1];
        List<SourceMetadata.Field> fs = new ArrayList<>();
        String escape = m.getSearchStringEscape();
        try (ResultSet rs =
            m.getColumns(
                c.getCatalog(),
                schema.replace("_", escape + "_"),
                table.replace("_", escape + "_"),
                null)) {
          while (rs.next())
            if (schema.equals(rs.getString("TABLE_SCHEM"))
                && table.equals(rs.getString("TABLE_NAME")))
              fs.add(
                  new SourceMetadata.Field(
                      rs.getString("COLUMN_NAME"),
                      rs.getString("TYPE_NAME"),
                      rs.getInt("NULLABLE") != DatabaseMetaData.columnNoNulls));
        }
        if (fs.isEmpty()) continue;
        // 零行读取实际验证 SELECT 权限；元数据可见性不能冒充读取权限。
        try (Statement st = c.createStatement()) {
          st.setQueryTimeout(3);
          st.executeQuery("SELECT * FROM " + QueryPlanner.quote(r) + " WHERE 1=0").close();
        }
        Map<Short, String> pk = new TreeMap<>();
        try (ResultSet rs = m.getPrimaryKeys(c.getCatalog(), schema, table)) {
          while (rs.next()) pk.put(rs.getShort("KEY_SEQ"), rs.getString("COLUMN_NAME"));
        }
        List<SourceMetadata.Relation> relations = new ArrayList<>();
        try (ResultSet rs = m.getImportedKeys(c.getCatalog(), schema, table)) {
          while (rs.next()) {
            String target = rs.getString("PKTABLE_SCHEM") + "." + rs.getString("PKTABLE_NAME");
            if (s.accessScope().contains(target))
              relations.add(
                  new SourceMetadata.Relation(
                      rs.getString("FKCOLUMN_NAME"), target, rs.getString("PKCOLUMN_NAME")));
          }
        }
        List<SourceMetadata.Index> indexes = new ArrayList<>();
        try (ResultSet rs = m.getIndexInfo(c.getCatalog(), schema, table, false, false)) {
          while (rs.next())
            if (rs.getString("INDEX_NAME") != null && rs.getString("COLUMN_NAME") != null)
              indexes.add(
                  new SourceMetadata.Index(
                      rs.getString("INDEX_NAME"),
                      List.of(rs.getString("COLUMN_NAME")),
                      !rs.getBoolean("NON_UNIQUE")));
        }
        out.add(
            new SourceMetadata.Resource(r, fs, new ArrayList<>(pk.values()), relations, indexes));
      }
      return new SourceMetadata(out);
    }
  }

  /** ClickHouse 的 database 是 catalog；独立读取系统目录，不能套用 PG 的 schema 行为。 */
  SourceMetadata clickHouseMetadata(Connection c, SourceSpec s) throws Exception {
    List<SourceMetadata.Resource> out = new ArrayList<>();
    for (String r : s.accessScope()) {
      QueryPlanner.identifier(r);
      String[] parts = r.split("\\.");
      if (parts.length != 2) throw QueryPlanner.problem("INVALID_RESOURCE");
      try (PreparedStatement engine =
          c.prepareStatement("SELECT engine FROM system.tables WHERE database=? AND name=?")) {
        engine.setString(1, parts[0]);
        engine.setString(2, parts[1]);
        engine.setQueryTimeout(3);
        try (ResultSet rs = engine.executeQuery()) {
          if (!rs.next()) continue;
          if (!rs.getString(1).endsWith("MergeTree"))
            throw QueryPlanner.problem("CLICKHOUSE_MERGETREE_REQUIRED");
        }
      }
      List<SourceMetadata.Field> fs = new ArrayList<>();
      List<String> sorting = new ArrayList<>();
      try (PreparedStatement columns =
          c.prepareStatement(
              "SELECT name,type,is_in_primary_key FROM system.columns WHERE database=? AND table=?"
                  + " ORDER BY position")) {
        columns.setString(1, parts[0]);
        columns.setString(2, parts[1]);
        columns.setQueryTimeout(3);
        try (ResultSet rs = columns.executeQuery()) {
          while (rs.next()) {
            String name = rs.getString(1), type = rs.getString(2);
            fs.add(new SourceMetadata.Field(name, type, type.startsWith("Nullable(")));
            if (rs.getBoolean(3)) sorting.add(name);
          }
        }
      }
      // MergeTree 排序键不保证唯一，证据改用物理分片和行偏移。
      fs.add(new SourceMetadata.Field("_part", "String", false));
      fs.add(new SourceMetadata.Field("_part_offset", "UInt64", false));
      try (Statement st = c.createStatement()) {
        st.setQueryTimeout(3);
        st.executeQuery("SELECT _part,_part_offset FROM " + QueryPlanner.quote(r) + " WHERE 1=0")
            .close();
      }
      out.add(
          new SourceMetadata.Resource(
              r,
              fs,
              List.of("_part", "_part_offset"),
              List.of(),
              sorting.isEmpty()
                  ? List.of()
                  : List.of(new SourceMetadata.Index("sorting_key", sorting, false))));
    }
    return new SourceMetadata(out);
  }

  @Override
  QueryResult execute(SourceSpec s, QuerySpec q) throws Exception {
    Set<String> used = new LinkedHashSet<>();
    used.add(q.resource());
    q.joins().forEach(j -> used.add(j.resource()));
    for (String resource : used) {
      exact(s, resource);
      QueryPlanner.identifier(resource);
    }
    var metadata = readMetadata(new SourceSpec(s.id(), s.kind(), s.connection(), used));
    var plan = new QueryPlanner().plan(s, q, metadata);
    try (Connection c = open(s)) {
      boolean pg = s.kind().equals("RELATIONAL");
      if (pg) {

      }
      String sql =

              + (pg
                  ? ""

      try (PreparedStatement st = c.prepareStatement(sql)) {
        st.setQueryTimeout(Math.max(1, (plan.timeoutMs() + 999) / 1000));
        st.setMaxRows(Math.min(500, plan.limit()));
        for (int i = 0; i < plan.parameters().size(); i++)
          st.setObject(i + 1, plan.parameters().get(i));
        List<Map<String, Object>> rows = new ArrayList<>();
        List<QueryResult.SourceRecord> evidence = new ArrayList<>();
        List<SourceMetadata.Field> columns = new ArrayList<>();
        try (ResultSet rs = st.executeQuery()) {
          for (int i = 0; i < q.projection().size(); i++)
            columns.add(
                new SourceMetadata.Field(
                    q.projection().get(i), rs.getMetaData().getColumnTypeName(i + 1), true));
          while (rows.size() < plan.limit() && rs.next()) {
            Map<String, Object> row = new LinkedHashMap<>();
            for (int i = 0; i < q.projection().size(); i++)
              row.put(q.projection().get(i), rs.getObject(i + 1));
            rows.add(row);
            int index = q.projection().size() + 1;
            for (var entry : plan.keys().entrySet()) {
              Map<String, Object> key = new LinkedHashMap<>();
              for (String k : entry.getValue()) key.put(k, rs.getObject(index++));
              evidence.add(new QueryResult.SourceRecord(s.id(), entry.getKey(), key));
            }
          }
        } finally {
          if (pg) c.rollback();
        }
        return new QueryResult(columns, rows, evidence);
      }
    }
  }

/** 仅精确白名单端点，禁止所有重定向；分页声明由受信配置提供。 */
public final class HttpSourceAdapter extends AdapterSupport {
  private final HttpClient client =
      HttpClient.newBuilder()
          .connectTimeout(Duration.ofSeconds(2))
          .followRedirects(HttpClient.Redirect.NEVER)
          .build();

  @Override
  public Set<String> capabilities() {
    return Set.of(
        "GET",
        "POST_JSON_TEMPLATE",
        "EXACT_ENDPOINT",
        "DECLARED_PAGE_NUMBER",
        "FILTER_BEFORE_LIMIT",
        "NO_REDIRECT",
        "MAX_16_MIB_RESPONSE");
  }

  URI endpoint(SourceSpec s, String r, int page) {
    exact(s, r);
    boolean declared = false;
    for (JsonNode node : s.connection().path("endpoints"))
      if (node.isTextual() && node.textValue().equals(r)) declared = true;
    if (!declared || !r.matches("/[A-Za-z0-9_/-]*") || r.contains(".."))
      throw QueryPlanner.problem("HTTP_ENDPOINT_REJECTED");
    URI base = URI.create(text(s, "baseUrl"));
    if (!Set.of("http", "https").contains(base.getScheme())
        || base.getHost() == null
        || base.getUserInfo() != null
        || base.getQuery() != null
        || base.getFragment() != null
        || !(base.getPath().isEmpty() || base.getPath().equals("/")))
      throw QueryPlanner.problem("INVALID_HTTP_BASE");
    String pageParam = s.connection().path("pageParam").asText("");
    if (!pageParam.isEmpty()) QueryPlanner.identifier(pageParam);
    return base.resolve(r + (pageParam.isEmpty() ? "" : "?" + pageParam + "=" + page));
  }

  JsonNode page(SourceSpec s, String r, int page, int timeout) throws Exception {
    HttpRequest.Builder request =
        HttpRequest.newBuilder(endpoint(s, r, page))
            .timeout(Duration.ofMillis(Math.min(timeout, 10000)));
    var headers = s.connection().path("headers");
    if (!headers.isMissingNode() && !headers.isObject())
      throw QueryPlanner.problem("INVALID_HTTP_HEADERS");
    headers
        .fields()
        .forEachRemaining(
            e -> {
              if (!e.getValue().isTextual()
                  || Set.of("host", "connection", "content-length")
                      .contains(e.getKey().toLowerCase(Locale.ROOT)))
                throw QueryPlanner.problem("INVALID_HTTP_HEADERS");
              request.header(e.getKey(), e.getValue().textValue());
            });
    String method = s.connection().path("method").asText("GET");
    if (method.equals("GET")) request.GET();
    else if (method.equals("POST")) {
      JsonNode body = s.connection().path("body");
      if (!body.isObject()) throw QueryPlanner.problem("INVALID_HTTP_TEMPLATE");
      var payload = body.deepCopy();
      if (!s.connection().path("pageParam").asText("").isEmpty())
        ((com.fasterxml.jackson.databind.node.ObjectNode) payload)
            .put(s.connection().path("pageParam").asText(), page);
      request
          .header("Content-Type", "application/json")
          .POST(HttpRequest.BodyPublishers.ofString(payload.toString()));
    } else throw QueryPlanner.problem("HTTP_METHOD_REJECTED");
    // sendAsync 的完整 body future 同时受截止时间限制，不能仅对响应头计时。
    var future = client.sendAsync(request.build(), info -> new BoundedBody());
    HttpResponse<byte[]> response;
    try {
      response = future.get(Math.min(timeout, 10000), java.util.concurrent.TimeUnit.MILLISECONDS);
    } catch (java.util.concurrent.TimeoutException e) {
      future.cancel(true);
      throw QueryPlanner.problem("QUERY_TIMEOUT");
    }
    if (response.statusCode() == 401 || response.statusCode() == 403)
      throw QueryPlanner.problem("AUTHENTICATION_FAILED");
    if (response.statusCode() < 200 || response.statusCode() >= 300)
      throw QueryPlanner.problem("HTTP_RESPONSE_REJECTED");
    JsonNode data = JSON.readTree(response.body());
    if (!data.isArray()) throw QueryPlanner.problem("INVALID_HTTP_RESPONSE");
    return data;
  }

  private static final class BoundedBody implements HttpResponse.BodySubscriber<byte[]> {
    private final ByteArrayOutputStream bytes = new ByteArrayOutputStream();
    private final java.util.concurrent.CompletableFuture<byte[]> result =
        new java.util.concurrent.CompletableFuture<>();
    private java.util.concurrent.Flow.Subscription subscription;

    @Override
    public java.util.concurrent.CompletionStage<byte[]> getBody() {
      return result;
    }

    @Override
    public void onSubscribe(java.util.concurrent.Flow.Subscription subscription) {
      this.subscription = subscription;
      subscription.request(1);
    }

    @Override
    public void onNext(List<java.nio.ByteBuffer> chunks) {
      for (var chunk : chunks) {
        if (chunk.remaining() > MAX_BYTES - bytes.size()) {
          subscription.cancel();
          result.completeExceptionally(QueryPlanner.problem("RESPONSE_TOO_LARGE"));
          return;
        }
        byte[] data = new byte[chunk.remaining()];
        chunk.get(data);
        bytes.writeBytes(data);
      }
      subscription.request(1);
    }

    @Override
    public void onError(Throwable failure) {
      result.completeExceptionally(failure);
    }

    @Override
    public void onComplete() {
      result.complete(bytes.toByteArray());
    }
  }

  @Override
  void connectivity(SourceSpec s) throws Exception {
    uriSocket(text(s, "baseUrl"), text(s, "baseUrl").startsWith("https:") ? 443 : 80);
  }

  @Override
  void authenticate(SourceSpec s) throws Exception {
    if (s.accessScope().isEmpty()) throw QueryPlanner.problem("SCOPE_EXCEEDED");
    page(s, s.accessScope().iterator().next(), 1, 3000);
  }

  @Override
  SourceMetadata readMetadata(SourceSpec s) throws Exception {
    List<SourceMetadata.Resource> out = new ArrayList<>();
    for (String r : s.accessScope()) {
      var data = page(s, r, 1, 3000);
      List<Map<String, Object>> rows = new ArrayList<>();
      for (var item : data) {
        if (!item.isObject()) throw QueryPlanner.problem("INVALID_HTTP_RESPONSE");
        rows.add(
            JSON.convertValue(
                item, new com.fasterxml.jackson.core.type.TypeReference<Map<String, Object>>() {}));
      }
      out.add(inferred(r, rows));
    }
    return new SourceMetadata(out);
  }

  @Override
  QueryResult execute(SourceSpec s, QuerySpec q) throws Exception {
    flat(q);
    endpoint(s, q.resource(), 1);
    long started = System.nanoTime();
    int pageSize = number(s, "pageSize", 100), maxPages = number(s, "maxPages", 1);
    if (pageSize < 1 || pageSize > 1000 || maxPages < 1 || maxPages > 100)
      throw QueryPlanner.problem("INVALID_HTTP_PAGINATION");
    List<Map<String, Object>> out = new ArrayList<>();
    List<QueryResult.SourceRecord> evidence = new ArrayList<>();
    SourceMetadata.Resource meta = null;
    for (int page = 1; page <= maxPages; page++) {
      deadline(started, q);
      var data =
          page(
              s,
              q.resource(),
              page,
              Math.max(1, q.timeoutMs() - (int) ((System.nanoTime() - started) / 1_000_000)));
      if (data.size() > pageSize) throw QueryPlanner.problem("HTTP_PAGE_SIZE_EXCEEDED");
      int index = 0;
      for (var item : data) {
        if (!item.isObject()) throw QueryPlanner.problem("INVALID_HTTP_RESPONSE");
        Map<String, Object> row =
            JSON.convertValue(
                item, new com.fasterxml.jackson.core.type.TypeReference<Map<String, Object>>() {});
        if (meta == null) {
          meta = inferred(q.resource(), List.of(row));
          fields(q, row.keySet());
        }
        if (matches(q, row)) {
          out.add(project(q, row));
          evidence.add(
              new QueryResult.SourceRecord(
                  s.id(), q.resource(), Map.of("page", page, "index", index)));
          if (out.size() >= q.limit()) return new QueryResult(meta.fields(), out, evidence);
        }
        index++;
      }
      if (data.size() < pageSize || s.connection().path("pageParam").asText("").isEmpty())
        return new QueryResult(meta == null ? List.of() : meta.fields(), out, evidence);
    }
    throw QueryPlanner.problem("HTTP_PAGINATION_LIMIT");
  }
}

  TableShape table(CqlSession c, SourceSpec s, String r) {

    QueryPlanner.identifier(r);
    String[] parts = r.split("\\.");
    if (parts.length != 2 || !parts[0].equals(text(s, "keyspace")))
      throw QueryPlanner.problem("SCOPE_EXCEEDED");

    Map<String, String> columns = new LinkedHashMap<>();

    var rows =
        c.execute(

                        + " keyspace_name=? AND table_name=?",
                    parts[0],

      String name = row.getString("column_name"),
          type = row.getString("type"),
          kind = row.getString("kind");
      columns.put(name, type);
      if ("partition_key".equals(kind)) partition.put(row.getInt("position"), name);
      if ("clustering".equals(kind)) clustering.put(row.getInt("position"), name);
    }
    if (columns.isEmpty()) throw QueryPlanner.problem("RESOURCE_NOT_FOUND");
    List<String> primary = new ArrayList<>(partition.values());
    primary.addAll(clustering.values());
    List<SourceMetadata.Index> indexes = new ArrayList<>();
    var indexRows =
        c.execute(
            SimpleStatement.newInstance(
                    "SELECT index_name,options FROM system_schema.indexes WHERE keyspace_name=? AND"
                        + " table_name=?",
                    parts[0],
                    parts[1])
                .setTimeout(Duration.ofSeconds(3)));

      indexes.add(

      all.addAll(table.columns().keySet());

      partition.addAll(table.partition());
      Map<String, com.fasterxml.jackson.databind.JsonNode> equal = new LinkedHashMap<>();
      for (var f : q.filters()) {
        if (f.operator() != QuerySpec.Operator.EQ

            || equal.putIfAbsent(f.column(), f.values().getFirst()) != null)

      for (var guard : q.guardBindings().entrySet()) {
        if (!partition.contains(guard.getValue()))
          throw QueryPlanner.problem("CASSANDRA_GUARD_PARTITION_REQUIRED");
        if (!q.scope().allows(guard.getKey(), equal.get(guard.getValue())))
          return new QueryResult(List.of(), List.of(), List.of());
      }

      projection.addAll(table.primary());

          projection.stream()

          partition.stream()

              .collect(java.util.stream.Collectors.joining(" AND "));
      var prepared =
          c.prepare(
              "SELECT "
                  + select
                  + " FROM "
                  + QueryPlanner.quote(q.resource())
                  + " WHERE "
                  + where
                  + " LIMIT "
                  + q.limit());
      Object[] values =
          partition.stream()
              .map(
                  k -> {
                    var n = equal.get(k);
                    var type = table.columns().get(k);
                    return switch (type) {
                      case "int" -> {
                        if (!n.isIntegralNumber() || !n.canConvertToInt())
                          throw QueryPlanner.problem("FILTER_TYPE_MISMATCH");
                        yield n.intValue();
                      }
                      case "bigint" -> {
                        if (!n.isIntegralNumber() || !n.canConvertToLong())
                          throw QueryPlanner.problem("FILTER_TYPE_MISMATCH");
                        yield n.longValue();
                      }
                      case "uuid" -> {
                        if (!n.isTextual()) throw QueryPlanner.problem("FILTER_TYPE_MISMATCH");
                        yield UUID.fromString(n.textValue());
                      }
                      case "boolean" -> {
                        if (!n.isBoolean()) throw QueryPlanner.problem("FILTER_TYPE_MISMATCH");
                        yield n.booleanValue();
                      }
                      case "text", "varchar", "ascii" -> {
                        if (!n.isTextual()) throw QueryPlanner.problem("FILTER_TYPE_MISMATCH");
                        yield n.textValue();
                      }
                      default -> throw QueryPlanner.problem("CASSANDRA_PARTITION_TYPE_UNSUPPORTED");
                    };
                  })
              .toArray();
      var result =
          c.execute(
              prepared
                  .bind(values)
                  .setTimeout(Duration.ofMillis(q.timeoutMs()))
                  .setPageSize(q.limit()));
      List<Map<String, Object>> out = new ArrayList<>();
      List<QueryResult.SourceRecord> evidence = new ArrayList<>();
      for (var row : result) {
        Map<String, Object> data = new LinkedHashMap<>();
        for (String p : q.projection()) data.put(p, row.getObject(p));

        Map<String, Object> key = new LinkedHashMap<>();

        evidence.add(new QueryResult.SourceRecord(s.id(), q.resource(), key));
        if (out.size() >= q.limit()) break;
      }
      return new QueryResult(inferred(q.resource(), out).fields(), out, evidence);
    }
  }

      throw QueryPlanner.problem("GUARD_ATTR_UNRESOLVED");

      return execute(s, q);
    } catch (Exception e) {

    }
  }

      if (!q.scope().allows(b.getKey(), JSON.valueToTree(row.get(b.getValue())))) return false;
    for (var f : q.filters()) {
      JsonNode actual = JSON.valueToTree(row.get(f.column()));
      if (actual == null || actual.isNull()) return false;
      if (f.operator() == QuerySpec.Operator.IN) {
        if (f.values().stream().noneMatch(v -> compare(actual, v) == 0)) return false;
      } else {
        int c = compare(actual, f.values().getFirst());

          case EQ -> c == 0;
          case GT -> c > 0;
          case GE -> c >= 0;
          case LT -> c < 0;
          case LE -> c <= 0;
          default -> false;
        }) return false;
      }
    }
    return true;
  }

  static int compare(JsonNode a, JsonNode b) {
    if (a.isNumber() && b.isNumber()) return a.decimalValue().compareTo(b.decimalValue());
    if (a.isTextual() && b.isTextual()) return a.textValue().compareTo(b.textValue());
    if (a.isBoolean() && b.isBoolean()) return Boolean.compare(a.booleanValue(), b.booleanValue());
    throw QueryPlanner.problem("FILTER_TYPE_MISMATCH");
  }

  static void fields(QuerySpec q, Set<String> fields) {
    Set<String> required = new HashSet<>(q.projection());
    required.addAll(q.guardBindings().values());
    q.filters().forEach(f -> required.add(f.column()));
    if (!fields.containsAll(required)) throw QueryPlanner.problem("UNKNOWN_COLUMN");
  }

  static Map<String, Object> project(QuerySpec q, Map<String, Object> row) {
    Map<String, Object> out = new LinkedHashMap<>();
    q.projection().forEach(c -> out.put(c, row.get(c)));
    return out;
  }

  static SourceMetadata.Resource inferred(String name, List<Map<String, Object>> rows) {
    Map<String, String> fields = new LinkedHashMap<>();
    for (var r : rows)
      r.forEach(
          (k, v) ->
              fields.putIfAbsent(
                  k,
                  v == null
                      ? "UNKNOWN"
                      : v instanceof Number
                          ? "NUMBER"
                          : v instanceof Boolean ? "BOOLEAN" : "STRING"));
    return new SourceMetadata.Resource(
        name,
        fields.entrySet().stream()
            .map(e -> new SourceMetadata.Field(e.getKey(), e.getValue(), true))
            .toList(),
        List.of(),
        List.of(),
        List.of());
  }

  static void deadline(long started, QuerySpec q) {
    if (System.nanoTime() - started > q.timeoutMs() * 1_000_000L)
      throw QueryPlanner.problem("QUERY_TIMEOUT");
  }

package cn.ontorepo.data.domain;

import com.fasterxml.jackson.databind.JsonNode;
import java.util.*;
import java.util.stream.Collectors;

/** 全部源表和列先校验，标签参数化并在 LIMIT 前执行。 */
public final class QueryPlanner {
  public record SqlPlan(
      String sql,
      List<Object> parameters,
      int limit,
      int timeoutMs,
      List<String> projection,
      Map<String, List<String>> keys) {}

  public static void identifier(String name) {
    if (name == null
        || !name.matches("[A-Za-z_][A-Za-z0-9_]{0,79}(\\.[A-Za-z_][A-Za-z0-9_]{0,79}){0,2}"))
      throw problem("INVALID_IDENTIFIER");
  }

  public static DataProblem problem(String code) {
    return new DataProblem(code, "数据源查询不满足访问边界或能力约束");
  }

  public static Object scalar(JsonNode n) {
    if (n == null || n.isNull() || n.isContainerNode()) throw problem("INVALID_FILTER");
    if (n.isNumber()) return n.decimalValue();
    if (n.isBoolean()) return n.booleanValue();
    return n.textValue();
  }

  public SqlPlan plan(SourceSpec source, QuerySpec q, SourceMetadata metadata) {
    if (!q.options().isEmpty()) throw problem("UNSUPPORTED_OPTION");
    Map<String, SourceMetadata.Resource> resources =
        metadata.resources().stream()
            .collect(Collectors.toMap(SourceMetadata.Resource::name, r -> r));
    Set<String> used = new LinkedHashSet<>();
    used.add(q.resource());
    q.joins().forEach(j -> used.add(j.resource()));
    for (String r : used) {
      identifier(r);
      if (!source.accessScope().contains(r)) throw problem("SCOPE_EXCEEDED");
      if (!resources.containsKey(r)) throw problem("RESOURCE_NOT_FOUND");
    }
    Set<String> columns = new HashSet<>();
    used.forEach(
        r ->
            resources
                .get(r)
                .fields()
                .forEach(
                    f -> {
                      identifier(f.name());
                      columns.add(r + "." + f.name());
                    }));
    java.util.function.Function<String, String> col =
        c -> {
          identifier(c);
          if (!columns.contains(c)) throw problem("UNKNOWN_COLUMN");
          return quote(c);
        };
    if (q.guardBindings().isEmpty()
        || !q.guardBindings().keySet().stream()
            .map(QueryPlanner::guardAttribute)
            .collect(Collectors.toSet())
            .containsAll(q.scope().clauses().keySet())) throw problem("GUARD_ATTR_UNRESOLVED");
    // JOIN 每一端都需分类标签；不能通过已授权任务关联出未分类算法。
    for (String resource : used)
      if (q.guardBindings().values().stream().noneMatch(c -> c.startsWith(resource + ".")))
        throw problem("GUARD_ATTR_UNRESOLVED");
    List<Object> params = new ArrayList<>();
    List<String> where = new ArrayList<>();
    for (var e : q.guardBindings().entrySet()) {
      String c = col.apply(e.getValue());
      where.add(c + " IS NOT NULL");
      var field =
          resources.get(e.getValue().substring(0, e.getValue().lastIndexOf('.'))).fields().stream()
              .filter(
                  f -> f.name().equals(e.getValue().substring(e.getValue().lastIndexOf('.') + 1)))
              .findFirst()
              .orElseThrow();
      if (field.type().toUpperCase(Locale.ROOT).matches(".*(CHAR|TEXT|STRING).*"))
        where.add("TRIM(" + c + ") <> ''");
      var values = q.scope().parameters().get(guardAttribute(e.getKey()));
      if (values != null) in(c, values, where, params);
    }
    for (var f : q.filters()) {
      String c = col.apply(f.column());
      if (f.operator() == null || f.values().size() > 1000) throw problem("INVALID_FILTER");
      if (f.operator() == QuerySpec.Operator.IN) in(c, f.values(), where, params);
      else {
        if (f.values().size() != 1) throw problem("INVALID_FILTER");
        where.add(
            c
                + " "
                + switch (f.operator()) {
                  case EQ -> "=";
                  case GT -> ">";
                  case GE -> ">=";
                  case LT -> "<";
                  case LE -> "<=";
                  default -> throw problem("INVALID_FILTER");
                }
                + " ?");
        params.add(scalar(f.values().getFirst()));
      }
    }
    String select = q.projection().stream().map(col).collect(Collectors.joining(","));
    Map<String, List<String>> keys = new LinkedHashMap<>();
    int keyIndex = 0;
    for (String r : used) {
      if (resources.get(r).primaryKey().isEmpty()) throw problem("SOURCE_KEY_REQUIRED");
      keys.put(r, resources.get(r).primaryKey());
      for (String key : resources.get(r).primaryKey())
        select += "," + col.apply(r + "." + key) + " AS \"__e" + (keyIndex++) + "\"";
    }
    StringBuilder sql =
        new StringBuilder("SELECT ").append(select).append(" FROM ").append(quote(q.resource()));
    Set<String> joined = new HashSet<>();
    joined.add(q.resource());
    for (var j : q.joins()) {
      String l = j.leftColumn(), r = j.rightColumn();
      if (!r.startsWith(j.resource() + ".")
          || joined.stream().noneMatch(x -> l.startsWith(x + "."))
          || !joined.add(j.resource())) throw problem("INVALID_JOIN");
      sql.append(" JOIN ")
          .append(quote(j.resource()))
          .append(" ON ")
          .append(col.apply(l))
          .append(" = ")
          .append(col.apply(r));
    }
    sql.append(" WHERE ").append(String.join(" AND ", where)).append(" LIMIT ?");
    params.add(q.limit());
    return new SqlPlan(
        sql.toString(),
        List.copyOf(params),
        q.limit(),
        q.timeoutMs(),
        q.projection(),
        Collections.unmodifiableMap(new LinkedHashMap<>(keys)));
  }

  private static void in(
      String col, List<JsonNode> values, List<String> where, List<Object> params) {
    if (values.isEmpty()) {
      where.add("1=0");
      return;
    }
    where.add(col + " IN (" + String.join(",", Collections.nCopies(values.size(), "?")) + ")");
    values.forEach(v -> params.add(scalar(v)));
  }

  public static String quote(String value) {
    identifier(value);
    return Arrays.stream(value.split("\\."))
        .map(x -> "\"" + x + "\"")
        .collect(Collectors.joining("."));
  }

  /** 同一属性在 JOIN 多表绑定时使用 attr@resource，仍应用同一属性范围。 */
  public static String guardAttribute(String binding) {
    String[] parts = binding.split("@", -1);
    if (parts.length > 2 || !parts[0].matches("[A-Za-z][A-Za-z0-9_]{0,79}"))
      throw problem("GUARD_ATTR_UNRESOLVED");
    if (parts.length == 2) identifier(parts[1]);
    return parts[0];
  }
}

package cn.ontorepo.data.domain;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.node.DecimalNode;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

/**
 * 授权标签范围的不可变交集，不负责判断授权记录是否存在。
 *
 * <p>调用者必须先验证入口授权记录；记录存在且范围 null 才代表不限。 策略、端点、空间按属性逐项求交，任何层不能扩大另一层；同一空间范围应用两次保持等价。
 * 缺少分类标签始终拒绝，即使所有范围均为不限，也不能把未分类对象当作公共数据。
 */
public record LabelScope(Map<String, Set<JsonNode>> clauses) {
  public LabelScope {
    if (clauses == null || clauses.size() > 64) throw invalid();
    Map<String, Set<JsonNode>> copy = new LinkedHashMap<>();
    clauses.forEach(
        (attribute, values) -> {
          if (attribute == null
              || !attribute.matches("[A-Za-z][A-Za-z0-9_]{0,79}")
              || values == null
              || values.size() > 1000) throw invalid();
          Set<JsonNode> normalized = new LinkedHashSet<>();
          for (JsonNode value : values) {
            if (!classified(value)) throw invalid();
            normalized.add(normalize(value));
          }
          copy.put(attribute, Set.copyOf(normalized));
        });
    clauses = Map.copyOf(copy);
  }

  public static LabelScope unrestricted() {
    return new LabelScope(Map.of());
  }

  /** 仅允许明确的集合或集合合取，不接受原始 SQL、通配符和任意表达式。 */
  public static LabelScope parse(JsonNode source) {
    return parse(source, 0);
  }

  private static LabelScope parse(JsonNode source, int depth) {
    if (source == null || source.isNull()) return unrestricted();
    if (!source.isObject() || depth > 16) throw invalid();
    if (source.size() == 1 && source.has("all")) {
      JsonNode all = source.get("all");
      if (!all.isArray() || all.isEmpty() || all.size() > 64) throw invalid();
      LabelScope combined = unrestricted();
      for (JsonNode child : all) {
        if (!child.isObject()) throw invalid();
        combined = intersect(combined, parse(child, depth + 1));
      }
      return combined;
    }
    JsonNode attr = source.get("attr"), values = source.get("in");
    if (source.size() != 2
        || attr == null
        || !attr.isTextual()
        || !attr.textValue().matches("[A-Za-z][A-Za-z0-9_]{0,79}")
        || values == null
        || !values.isArray()
        || values.size() > 1000) throw invalid();
    Set<JsonNode> allowed = new LinkedHashSet<>();
    for (JsonNode value : values) {
      if (!classified(value)) throw invalid();
      allowed.add(normalize(value));
    }
    return new LabelScope(Map.of(attr.textValue(), allowed));
  }

  public static LabelScope intersect(LabelScope... scopes) {
    Map<String, Set<JsonNode>> result = new LinkedHashMap<>();
    for (LabelScope scope : scopes) {
      if (scope == null) throw invalid();
      for (var entry : scope.clauses.entrySet()) {
        Set<JsonNode> existing = result.get(entry.getKey());
        if (existing == null) result.put(entry.getKey(), new LinkedHashSet<>(entry.getValue()));
        else existing.retainAll(entry.getValue());
      }
    }
    return new LabelScope(result);
  }

  /** SQL 计划器将空交集编译为恒假谓词，不能在空列表时省略 WHERE。 */
  public boolean deniesAll() {
    return clauses.values().stream().anyMatch(Set::isEmpty);
  }

  /** 只判断当前属性；整条记录必须对全部必需 guard_attrs 逐项调用。 */
  public boolean allows(String attribute, JsonNode label) {
    if (deniesAll() || !classified(label)) return false;
    Set<JsonNode> allowed = clauses.get(attribute);
    return allowed == null || allowed.contains(normalize(label));
  }

  /** 返回参数化计划所需的只读集合；不得将其中字符串拼接到 SQL。 */
  public Map<String, List<JsonNode>> parameters() {
    Map<String, List<JsonNode>> result = new LinkedHashMap<>();
    clauses.forEach((attribute, values) -> result.put(attribute, List.copyOf(values)));
    return Map.copyOf(result);
  }

  private static JsonNode normalize(JsonNode value) {
    return value.isNumber()
        ? DecimalNode.valueOf(value.decimalValue().stripTrailingZeros())
        : value;
  }

  private static boolean classified(JsonNode value) {
    return value != null
        && (value.isBoolean()
            || value.isNumber()
            || value.isTextual()
                && !value.textValue().isBlank()
                && value.textValue().length() <= 1000);
  }

  private static DataProblem invalid() {
    return new DataProblem("INVALID_DATA_SCOPE", "数据范围必须为明确的属性值集合或集合合取");
  }
}

package cn.ontorepo.data.domain;

import java.nio.charset.StandardCharsets;
import java.security.SecureRandom;
import java.util.Arrays;
import java.util.Base64;
import javax.crypto.Cipher;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;

/**
 * 数据源连接配置的 AES-256-GCM 加密边界。
 *
 * <p>主密钥必须由进程环境提供，不生成固定默认值；每次加密采用随机 nonce。 数据源稳定标识作为附加认证数据，防止把一个数据源的密文拷贝到另一行后获得访问能力。
 * 连接配置只在真实适配器调用前解密，管理 API 不返回明文。
 */
public final class CredentialCipher {
  private static final int MAX_CLEAR_BYTES = 65536;
  private static final SecureRandom RANDOM = new SecureRandom();
  private final SecretKeySpec key;

  public CredentialCipher(String base64Key) {
    try {
      byte[] bytes = Base64.getDecoder().decode(base64Key);
      if (bytes.length != 32) throw new IllegalArgumentException();
      key = new SecretKeySpec(bytes, "AES");
      Arrays.fill(bytes, (byte) 0);
    } catch (RuntimeException e) {
      throw new DataProblem("CREDENTIAL_KEY_INVALID", "数据源主密钥必须为 Base64 编码的 32 字节密钥");
    }
  }

  public String encrypt(String sourceId, String plaintext) {
    identity(sourceId);
    if (plaintext == null || plaintext.isEmpty())
      throw new DataProblem("CREDENTIAL_CONFIG_INVALID", "连接配置不能为空");
    byte[] clear = plaintext.getBytes(StandardCharsets.UTF_8);
    if (clear.length > MAX_CLEAR_BYTES)
      throw new DataProblem("CREDENTIAL_CONFIG_INVALID", "连接配置超过大小限制");
    byte[] nonce = new byte[12];
    RANDOM.nextBytes(nonce);
    try {
      byte[] encrypted = cipher(Cipher.ENCRYPT_MODE, nonce, sourceId).doFinal(clear);
      byte[] combined = new byte[nonce.length + encrypted.length];
      System.arraycopy(nonce, 0, combined, 0, nonce.length);
      System.arraycopy(encrypted, 0, combined, nonce.length, encrypted.length);
      return "v1." + Base64.getUrlEncoder().withoutPadding().encodeToString(combined);
    } catch (Exception e) {
      throw new DataProblem("CREDENTIAL_ENCRYPTION_FAILED", "连接配置加密失败");
    } finally {
      Arrays.fill(clear, (byte) 0);
    }
  }

  public String decrypt(String sourceId, String encoded) {
    identity(sourceId);
    try {
      if (encoded == null || !encoded.startsWith("v1.") || encoded.length() > 90000)
        throw new IllegalArgumentException();
      byte[] combined = Base64.getUrlDecoder().decode(encoded.substring(3));
      if (combined.length < 29 || combined.length > MAX_CLEAR_BYTES + 28)
        throw new IllegalArgumentException();
      byte[] nonce = Arrays.copyOfRange(combined, 0, 12);
      byte[] clear =
          cipher(Cipher.DECRYPT_MODE, nonce, sourceId).doFinal(combined, 12, combined.length - 12);
      try {
        return new String(clear, StandardCharsets.UTF_8);
      } finally {
        Arrays.fill(clear, (byte) 0);
      }
    } catch (Exception e) {
      throw new DataProblem("CREDENTIAL_INTEGRITY_FAILED", "连接配置密文无效或不属于此数据源");
    }
  }

  private Cipher cipher(int mode, byte[] nonce, String sourceId) throws Exception {
    Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
    cipher.init(mode, key, new GCMParameterSpec(128, nonce));
    cipher.updateAAD(("ontorepo:datasource:v1:" + sourceId).getBytes(StandardCharsets.UTF_8));
    return cipher;
  }

  private static void identity(String sourceId) {
    if (sourceId == null || sourceId.isBlank() || sourceId.length() > 200)
      throw new DataProblem("CREDENTIAL_CONTEXT_INVALID", "数据源标识无效");
  }
}

# 数据适配内核 API

本模块独立构建：Java 21，`mvn -B -f onto-data/pom.xml -DargLine=-Xmx512m test`。业务层负责认证、空间权限、快照完整性、连接密文解密，适配器仅接收校验后的服务端对象。不要把请求直接反序列化成已授权的 QuerySpec。

`domain.SourceSpec(id,kind,connection,accessScope)`：connection 为解密后的 JsonNode，不记录日志；toString 隐藏连接。kind 与 dialect 必须明确匹配。`domain.AdapterRegistry.adapter(source)` 返回 `domain.SourceAdapter`。adapter 的 `probe` 返回 CONNECTIVITY/AUTHENTICATION/SCOPE 三阶段，失败后阶段 NOT_RUN；每段是 `ProbeResult.Stage(name,status,code)`，由管理层转换HTTP契约字段。`metadata` 返回 SourceMetadata.resources；每项含 name、fields(name,type,nullable)、primaryKey、relations、indexes。无可获得关系/索引时为空，不能当作完整性肯定结论。

`domain.QuerySpec(resource,projection,filters,joins,limit,timeoutMs,options,scope,guardBindings)` 是唯一查询入口。projection 为列列表；Filter(column,Operator,values) 支持 EQ/IN/GT/GE/LT/LE；Join(resource,leftColumn,rightColumn) 为内等值连接。limit 默认/最大500，timeoutMs 默认/最大10000。JDBC 列采用完整 `schema.table.column`；其余协议为资源内字段名。所有路径/表/集合/标签必须在 accessScope 内。options 未声明时一律拒绝。scope 是 LabelScope.intersect 的最终结果；guardBindings 为模型快照中的必需属性→源字段映射，必须非空且覆盖全部 scope 属性。未分类值(null/空串)默认拒绝。授权范围过滤在 LIMIT 前完成，空交集恒假。

`QueryResult(columns,rows,evidence)` 保留值的 Java 类型（数值/布尔/null/日期等）；rows 每项键为投影列，evidence 中 SourceRecord(sourceId,resource,key) 为源记录定位。JDBC 自动额外读取主键且不暴露为业务列；没有主键拒绝查询。文件采用行号定位，HTTP 采用页号/页内序号定位。上层继续补充映射快照/元数据快照/候选或版本摘要。

JOIN 每个参与表必须有分类守卫。同一属性需要绑定多个表时，使用 `region@public.algorithms -> public.algorithms.region`，其中 `region` 仍引用有效范围的同一属性；主表可用 `region -> public.tasks.region`。证据顺序按每行、再按主表/连接表；不同表各有一条 SourceRecord。

## 连接字段

| kind | dialect | connection 与 accessScope |
|---|---|---|
| RELATIONAL | postgresql | url/user/password；scope=`schema.table` |
| WAREHOUSE | clickhouse | url/user/password；scope=`database.table` |
| LAKE | s3-parquet | endpoint/accessKey/secretKey/bucket；scope=对象前缀；resource=对象键 |
| FILE | csv/json/excel | root（批准目录）；scope=相对文件路径；resource=相对路径 |
| HTTP | json | baseUrl/headers/endpoints；可选method=GET/POST、pageParam/pageSize/maxPages；scope=端点路径 |
| DOCUMENT | mongodb | uri/database；scope=`database.collection` |
| KEY_VALUE | redis | host/port/password/database；scope=键前缀 |
| WIDE_COLUMN | cassandra | contactPoints/port/datacenter/keyspace/user/password；scope=`keyspace.table` |
| GRAPH | neo4j | uri/user/password/database；scope=节点标签 |

连接对象只允许服务端受控配置。HTTP baseUrl 和文件 root 不应由执行请求覆盖；JDBC URL 不得启用允许本机文件读取的驱动选项。配置、原始 URI、源异常响应不输出给客户端。

## 能力限制

- PostgreSQL：普通主键表，内等值 JOIN；所有列使用完整资源前缀，未绑定守卫在元数据访问前拒绝。
- ClickHouse：仅 MergeTree 家族（包含复制变体）；独立查询 system.tables/system.columns 且按 database/table 参数过滤。排序键不保证唯一，证据使用 `_part/_part_offset`；分片合并后物理证据位置可能失效，上层应同时归档查询事实与快照。无本机实例，尚待真实验收。
- S3：单对象最多16MiB、每前缀最多1000对象；Parquet 仅非重复基础类型列，EQ/IN 谓词下推与列裁剪，分类检查在输出上限之前；证据为对象键及物理 rowIndex。尚未支持嵌套、Decimal/INT96 等复杂类型，明确拒绝。无需接受客户端原始路径或SQL。
- 文件：单文件最多16MiB/10万行；CSV、JSON对象数组和 .xlsx 首工作表。公式单元格直接拒绝；不执行公式/宏，.xls/.xlsm 暂不支持。真实路径必须留在批准root之下，拒绝任意 `..` 和绝对路径。
- HTTP：baseUrl 必须是无凭据/查询参数的 origin；端点精确声明，仅GET或固定JSON体POST模板（页码参数自动绑定），未开放客户端任意请求参数。响应为JSON对象数组；pageParam 为页码参数，每页至多1000条、至多100页，页未耗尽而触及配置上限时明确拒绝。完整响应体受截止时间和16MiB上限约束；禁止全部重定向。
- MongoDB：仅原生 find，服务端受限过滤/投影/LIMIT；元数据字段来自首条样本并含索引，不能视为完整文档Schema，不接收聚合脚本。
- Redis：仅STRING内的JSON对象，键名自动增加 `_key` 字段；resource 是精确批准前缀，使用 `_key EQ` 可做精确键查，否则 SCAN 最多1000步/10万键，过滤在结果上限前，无JOIN。
- Cassandra：仅完整分区键EQ，守卫必须绑定同一分区键；不支持任意过滤、ALLOW FILTERING 或JOIN。分区键仅 int/bigint/uuid/boolean/text/varchar/ascii，其他类型明确拒绝。
- Neo4j：节点属性查询；或 options 同时声明 `relationship`（关系类型标识符）、`targetLabel`（同样须在accessScope）时查询单跳有向关系。关系模板对两端使用同名分类属性和同一有效范围，可投影 `_targetId`、`_relationshipId`；关系模板暂不支持额外filters。节点与关系元素ID进入证据；其他Cypher/SQL JOIN不支持。

Windows 当前JDK的 Unix-domain socket 临时路径需要ASCII目录；本机测试命令使用 `-DargLine="-Xmx512m -Djdk.net.unixdomain.tmpdir=C:/Windows/Temp"`。这只影响测试运行参数，不改变协议授权规则。
